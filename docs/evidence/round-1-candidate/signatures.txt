RicciFlowSharpEstimate.Variational.existsUnique_capParameter (C : ℝ) (hC : 1 ≤ C) :
  ∃! q, 1 ≤ q ∧ Real.log q + 2 / 3 * (q ^ 3 - 1) = Real.log C
RicciFlowSharpEstimate.Variational.capParameter_spec (C : ℝ) (hC : 1 ≤ C) :
  1 ≤ RicciFlowSharpEstimate.Variational.capParameter C ∧
    Real.log (RicciFlowSharpEstimate.Variational.capParameter C) +
        2 / 3 * (RicciFlowSharpEstimate.Variational.capParameter C ^ 3 - 1) =

RicciFlowSharpEstimate.Variational.capParameter_eq_one_iff (C : ℝ) (hC : 1 ≤ C) :
  RicciFlowSharpEstimate.Variational.capParameter C = 1 ↔ C = 1
def RicciFlowSharpEstimate.Variational.orderedTriangle : Set (ℝ × ℝ) :=
{p | 0 ≤ p.1 ∧ p.1 < p.2 ∧ p.2 ≤ 1}
def RicciFlowSharpEstimate.Variational.pairFunctional : (ℝ → ℝ) → ℝ :=
fun k ↦ ∫ (p : ℝ × ℝ) in RicciFlowSharpEstimate.Variational.orderedTriangle, 2 * p.2 * Real.exp (k p.1 - k p.2)
def RicciFlowSharpEstimate.Variational.pairMarginal : (ℝ → ℝ) → ℝ → ℝ :=
fun g v ↦
  (∫ (s : ℝ), RicciFlowSharpEstimate.Variational.pairWeight g (v, s)) -
    ∫ (t : ℝ), RicciFlowSharpEstimate.Variational.pairWeight g (t, v)
def RicciFlowSharpEstimate.Variational.pairRemainder : (ℝ → ℝ) → (ℝ → ℝ) → ℝ :=
fun k g ↦
  ∫ (p : ℝ × ℝ) in RicciFlowSharpEstimate.Variational.orderedTriangle,
    2 * p.2 * Real.exp (g p.1 - g p.2) *
      (Real.exp (k p.1 - g p.1 - (k p.2 - g p.2)) - 1 - (k p.1 - g p.1 - (k p.2 - g p.2)))
RicciFlowSharpEstimate.Variational.pairFunctional_sub_eq_integral_marginal (k g : ℝ → ℝ)
  (hk : ContinuousOn k (Set.Icc 0 1)) (hg : ContinuousOn g (Set.Icc 0 1)) :
  RicciFlowSharpEstimate.Variational.pairFunctional k - RicciFlowSharpEstimate.Variational.pairFunctional g =
    (∫ (v : ℝ), RicciFlowSharpEstimate.Variational.pairMarginal g v * (k v - g v)) +
      RicciFlowSharpEstimate.Variational.pairRemainder k g
RicciFlowSharpEstimate.Variational.pairRemainder_nonneg (k g : ℝ → ℝ) :
  0 ≤ RicciFlowSharpEstimate.Variational.pairRemainder k g

def RicciFlowSharpEstimate.Variational.lowerContact : ℝ → ℝ :=
fun q ↦ 1 / √(q ^ 4 + 2 * q)
def RicciFlowSharpEstimate.Variational.upperContact : ℝ → ℝ :=
fun q ↦ RicciFlowSharpEstimate.Variational.lowerContact q * q ^ 2
RicciFlowSharpEstimate.Variational.contact_bounds (q : ℝ) (hq : 1 ≤ q) :
  0 < RicciFlowSharpEstimate.Variational.lowerContact q ∧
    RicciFlowSharpEstimate.Variational.lowerContact q ≤ RicciFlowSharpEstimate.Variational.upperContact q ∧
      RicciFlowSharpEstimate.Variational.upperContact q < 1
RicciFlowSharpEstimate.Variational.one_sub_upperContact_sq (q : ℝ) (hq : 0 < q) :
  1 - RicciFlowSharpEstimate.Variational.upperContact q ^ 2 =
    2 * RicciFlowSharpEstimate.Variational.lowerContact q ^ 2 * q
def RicciFlowSharpEstimate.Variational.freeExponential : ℝ → ℝ → ℝ :=
fun a v ↦ √(v / a) * Real.exp (2 / 3 * (√(v / a) ^ 3 - 1))
def RicciFlowSharpEstimate.Variational.freeLeftPrimitive : ℝ → ℝ → ℝ :=
fun a v ↦ a * Real.exp (2 / 3 * (√(v / a) ^ 3 - 1))
def RicciFlowSharpEstimate.Variational.freeRightPrimitive : ℝ → ℝ → ℝ :=
fun a v ↦ -2 * a ^ 2 * Real.exp (-(2 / 3 * (√(v / a) ^ 3 - 1)))
RicciFlowSharpEstimate.Variational.integral_freeExponential (a c d : ℝ) (ha : 0 < a) (hc : 0 < c) (hd : 0 < d) :
  ∫ (x : ℝ) in c..d, RicciFlowSharpEstimate.Variational.freeExponential a x =
    RicciFlowSharpEstimate.Variational.freeLeftPrimitive a d - RicciFlowSharpEstimate.Variational.freeLeftPrimitive a c
RicciFlowSharpEstimate.Variational.integral_two_mul_div_freeExponential (a c d : ℝ) (ha : 0 < a) (hc : 0 < c)
  (hd : 0 < d) :
  ∫ (x : ℝ) in c..d, 2 * x / RicciFlowSharpEstimate.Variational.freeExponential a x =
    RicciFlowSharpEstimate.Variational.freeRightPrimitive a d -
      RicciFlowSharpEstimate.Variational.freeRightPrimitive a c
def RicciFlowSharpEstimate.Variational.obstacleExponential : ℝ → ℝ → ℝ :=
fun q v ↦
  RicciFlowSharpEstimate.Variational.freeExponential (RicciFlowSharpEstimate.Variational.lowerContact q)
    (max (RicciFlowSharpEstimate.Variational.lowerContact q)
      (min v (RicciFlowSharpEstimate.Variational.upperContact q)))
def RicciFlowSharpEstimate.Variational.obstacleLogProfile : ℝ → ℝ → ℝ :=
fun C v ↦
  Real.log
    (RicciFlowSharpEstimate.Variational.obstacleExponential (RicciFlowSharpEstimate.Variational.capParameter C) v)
RicciFlowSharpEstimate.Variational.obstacleExponential_eq_piecewise (q v : ℝ) (hq : 1 ≤ q) :
  RicciFlowSharpEstimate.Variational.obstacleExponential q v =
    if v ≤ RicciFlowSharpEstimate.Variational.lowerContact q then 1
    else
      if v ≤ RicciFlowSharpEstimate.Variational.upperContact q then
        RicciFlowSharpEstimate.Variational.freeExponential (RicciFlowSharpEstimate.Variational.lowerContact q) v
      else Real.exp (RicciFlowSharpEstimate.Variational.logCap q)
RicciFlowSharpEstimate.Variational.obstacleLogProfile_admissible (C : ℝ) (hC : 1 ≤ C) :
  ContinuousOn (RicciFlowSharpEstimate.Variational.obstacleLogProfile C) (Set.Icc 0 1) ∧
    ∀ v ∈ Set.Icc 0 1,
      0 ≤ RicciFlowSharpEstimate.Variational.obstacleLogProfile C v ∧
        RicciFlowSharpEstimate.Variational.obstacleLogProfile C v ≤ Real.log C
RicciFlowSharpEstimate.Variational.obstacleLogProfile_one (v : ℝ) :
  RicciFlowSharpEstimate.Variational.obstacleLogProfile 1 v = 0
RicciFlowSharpEstimate.Variational.contact_separation_iff (C : ℝ) (hC : 1 ≤ C) :
  RicciFlowSharpEstimate.Variational.lowerContact (RicciFlowSharpEstimate.Variational.capParameter C) <
      RicciFlowSharpEstimate.Variational.upperContact (RicciFlowSharpEstimate.Variational.capParameter C) ↔
    1 < C
RicciFlowSharpEstimate.Analysis.exp_tangent_quadratic_lower (m x y : ℝ) (hx : m ≤ x) (hy : m ≤ y) :
  Real.exp m / 2 * (x - y) ^ 2 ≤ Real.exp x - Real.exp y - Real.exp y * (x - y)
