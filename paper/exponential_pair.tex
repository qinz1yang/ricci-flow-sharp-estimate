\begin{abstract}
We determine the minimum of an exponential interaction over continuous
functions with prescribed lower and upper obstacles. The minimizer consists
of two constant pieces joined by an explicit free arc. Its value is given
for every cap, and an exact convex deficit identity proves uniqueness.
At every nondegenerate cap the minimizer has derivative jumps at both
contacts; every admissible function differentiable in the interior has
strictly larger value.
The deficit also controls the squared $L^2$ distance to the minimizer with
an explicit positive cap-dependent constant. Applying this estimate to two
hemispheres gives quantitative reflection symmetry without imposing symmetry
as a constraint.
For smooth sphere metrics obtained from positive balanced rotational profiles
and their diffeomorphic pullbacks, a complete Haar decomposition gives
nonnegativity and zero-remainder rigidity without pinching.
The variational solution determines the exact conformal-Killing curvature
threshold, with smooth sharpness and full equality classification at the safe threshold.
A local plateau variation, strict curvature contraction and exact-moment
smooth recovery give a genuine negative one-form below that threshold.
Consequently the unrestricted rotational threshold is strictly smaller.
\end{abstract}

\section{The constrained problem}

For $C\geq1$, consider real functions whose restrictions to the unit interval
are continuous, and let
\[
 \mathcal A_C=\{k:\R\to\R:k|_{[0,1]}\in C([0,1]),\quad
      0\leq k(v)\leq\log C\text{ for }v\in[0,1]\},
\]
\[
 \I(k)=\int_0^1\int_0^s 2s e^{k(v)-k(s)}\,dv\,ds.
\]
The competitors and the variational functional are real-valued. The interval
constraints include both endpoints. In particular $\mathcal A_1$ contains
only functions vanishing on $[0,1]$. Values outside this interval do not
affect the functional; uniqueness always means equality on $[0,1]$.

\begin{theorem}\label{thm:minimum}
For every $C\geq1$ there is a unique $q\geq1$ satisfying
\begin{equation}\label{eq:cap}
 \log q+\frac23(q^3-1)=\log C.
\end{equation}
Set
\begin{equation}\label{eq:contacts}
 \alpha=\frac1{\sqrt{q^4+2q}},\qquad \beta=\alpha q^2.
\end{equation}
Then $0<\alpha\leq\beta<1$, and the function $k_*=\log E_*$, where
\begin{equation}\label{eq:candidate}
 E_*(v)=
 \begin{cases}
 1,&v\leq\alpha,\\
 \displaystyle\sqrt{v/\alpha}\,
  \exp\!\left(\frac23\bigl((v/\alpha)^{3/2}-1\bigr)\right),
   &\alpha\leq v\leq\beta,\\
 C,&v\geq\beta,
 \end{cases}
\end{equation}
belongs to $\mathcal A_C$. For every $k\in\mathcal A_C$,
\begin{equation}\label{eq:minimum}
 \I(k)\geq\frac23-\beta+\frac1{3\beta},
\end{equation}
with equality if and only if $k=k_*$ on $[0,1]$.
At $C=1$ the profile is identically zero and the minimum is $2/3$.
\end{theorem}

The degeneracy in the middle interval is essential at cap one. No derivative
formula on a nonempty free interval is needed for that endpoint.

\begin{lemma}\label{lem:parameter}
The parameter in \eqref{eq:cap} exists uniquely and increases strictly with
$C\geq1$. It equals one precisely when $C=1$. The contacts satisfy
\begin{equation}\label{eq:matching}
 \beta^2=\frac{q^3}{q^3+2},\qquad
 1-\beta^2=2\alpha^2q,\qquad
 \frac\alpha q=\frac{1-\beta^2}{2\beta}.
\end{equation}
The contacts are distinct precisely when $C>1$.
\end{lemma}

\begin{proof}
The function $F(q)=\log q+\frac23(q^3-1)$ is continuous and strictly
increasing on $[1,\infty)$. It has $F(1)=0$ and $F(C)\geq\log C$
for $C\geq1$, so the intermediate value theorem on $[1,C]$ gives the
unique parameter. Strict increase of $F$ and of the logarithm gives the
claimed dependence on $C$. Squaring \eqref{eq:contacts} gives
\eqref{eq:matching}. In particular $\beta<1$, while $q\geq1$ gives
$\alpha\leq\beta$, strictly when $q>1$.
\end{proof}

The free expression in \eqref{eq:candidate} equals one at $v=\alpha$
and $q\exp(\frac23(q^3-1))=C$ at $v=\beta$. Both its square-root
and exponential factors are nonnegative and increasing for positive $v$.
It therefore joins the obstacles continuously and stays between their
values. This proves the asserted admissibility.

\section{Exact primitives and marginal}

Write $\phi(v)=\frac23((v/\alpha)^{3/2}-1)$ on the free arc, and define
the actual prefix and tail integrals
\[
 P(v)=\int_0^v E_*(t)\,dt,\qquad
 Q(v)=\int_v^1\frac{2s}{E_*(s)}\,ds.
\]
\begin{lemma}\label{lem:primitives}
The two integrals have the formulas
\begin{equation}\label{eq:primitives}
 P(v)=
 \begin{cases}
 v,&v\leq\alpha,\\
 \alpha e^{\phi(v)},&\alpha\leq v\leq\beta,\\
 C(v-\beta+\alpha/q),&v\geq\beta,
 \end{cases}
 \quad
 Q(v)=
 \begin{cases}
 3\alpha^2-v^2,&v\leq\alpha,\\
 2\alpha^2 e^{-\phi(v)},&\alpha\leq v\leq\beta,\\
 (1-v^2)/C,&v\geq\beta,
 \end{cases}
\end{equation}
for $v\in[0,1]$. Consequently the marginal density
$A(v)=E_*(v)Q(v)-2vP(v)/E_*(v)$ is
\begin{equation}\label{eq:marginal}
 A(v)=
 \begin{cases}
 3(\alpha^2-v^2),&0\leq v\leq\alpha,\\
 0,&\alpha\leq v\leq\beta,\\
 -3(v-\beta)\bigl(v+1/(3\beta)\bigr),&\beta\leq v\leq1.
 \end{cases}
\end{equation}
\end{lemma}

\begin{proof}
Direct differentiation on the positive half-line gives
\[
 \frac{d}{dv}\bigl(\alpha e^{\phi(v)}\bigr)=E_*(v),\qquad
 \frac{d}{dv}\bigl(-2\alpha^2 e^{-\phi(v)}\bigr)
 =\frac{2v}{E_*(v)}
\]
on the free arc. At its upper endpoint the first primitive has value
$C\alpha/q$, while the second has value $-(1-\beta^2)/C$ by
\eqref{eq:matching}. Splitting the integrals at the contacts gives
\eqref{eq:primitives}; the calculation also applies when the contacts
coincide. Substitution into the definition of $A$ proves the first two
pieces of \eqref{eq:marginal}. On the upper piece it gives
$1-3v^2+2v\beta-2v\alpha/q$, which has the stated factorization by
the last identity in \eqref{eq:matching}.
\end{proof}

\section{Convex calibration and rigidity}

For continuous $k,g$ on $[0,1]$, put $\delta=k-g$ and
\[
 A_g(v)=e^{g(v)}\int_v^1 2s e^{-g(s)}\,ds
       -2v e^{-g(v)}\int_0^v e^{g(t)}\,dt.
\]
\begin{lemma}\label{lem:deficit}
The exact identity
\begin{align}\label{eq:deficit}
 \I(k)-\I(g)
 &=\int_0^1 A_g(v)\delta(v)\,dv+\mathcal R(k,g),\\
 \mathcal R(k,g)
 &=\int_0^1\int_0^s 2s e^{g(v)-g(s)}
 \left[e^{\delta(v)-\delta(s)}-1-\delta(v)+\delta(s)\right]dv\,ds
 \nonumber
\end{align}
holds. The remainder is nonnegative, and it vanishes if and only if
$\delta$ is constant on $[0,1]$.
\end{lemma}

\begin{proof}
Apply the identity $e^{x+y}-e^x=e^xy+e^x(e^y-1-y)$ with
$x=g(v)-g(s)$ and $y=\delta(v)-\delta(s)$. Fubini's theorem identifies
the linear term with the stated marginal integral. All integrands are
continuous on the compact square before restriction to the triangle, so
these integrations are justified. The inequality $e^y\geq1+y$ proves
nonnegativity. If the remainder vanishes, its nonnegative integrand is
zero almost everywhere. Continuity makes it zero at every point of the
open triangle $0<v<s<1$: otherwise it would be positive on an open set
of positive measure. The prefactor is positive there, and strictness of
$e^y>1+y$ for $y\ne0$ gives $\delta(v)=\delta(s)$. Any two interior
points can be ordered, so $\delta$ is constant in the interior, and
continuity extends that constant to the endpoints. The converse follows
directly from the integrand.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:minimum}]
Take $g=k_*$. On the lower obstacle $\delta=k\geq0$ and $A\geq0$;
on the upper obstacle $\delta=k-\log C\leq0$ and $A\leq0$; on the
free arc $A=0$. Thus both terms in \eqref{eq:deficit} are nonnegative,
and $k_*$ is a minimizer. If equality holds, Lemma~\ref{lem:deficit}
gives $k-k_*=c$ on the interval. At zero the lower constraint gives
$c\geq0$, while at one the upper constraint gives $c\leq0$. Therefore
$c=0$.

It remains to evaluate the value. Fubini gives
$\I(k_*)=\int_0^1 2vP(v)/E_*(v)\,dv$. The integrand on the three
pieces is, respectively,
\[
 2v^2,\qquad 2\alpha^2\sqrt{v/\alpha},\qquad
 2v(v-\beta+\alpha/q).
\]
Its integral is consequently
\[
 \frac23\alpha^3+\frac43\alpha^3(q^3-1)
 +\frac23(1-\beta^3)-(\beta-\alpha/q)(1-\beta^2).
\]
The identities \eqref{eq:matching} reduce this expression to
$\frac23-\beta+1/(3\beta)$. The integration of the middle term is
valid also on a degenerate interval. Alternatively, at $C=1$ one
directly has $\I(0)=\int_0^1 2s^2\,ds=2/3$.
\end{proof}

\begin{theorem}\label{thm:contacts}
For $C>1$, the one-sided derivatives of the minimizer are
\[
 k'_{*,-}(\alpha)=0,\qquad k'_{*,+}(\alpha)=\frac3{2\alpha},
 \qquad
 k'_{*,-}(\beta)=\frac1{2\beta}+\frac q\alpha,
 \qquad k'_{*,+}(\beta)=0.
\]
In particular, every $k\in\mathcal A_C$ differentiable on $(0,1)$
satisfies strict inequality in \eqref{eq:minimum}.
\end{theorem}

\begin{proof}
The obstacles are constant. In the free arc direct differentiation gives
$k_*'(v)=1/(2v)+\sqrt{v/\alpha}/\alpha$. Since $\alpha<\beta$, its
one-sided limits give the displayed positive derivatives. Both contacts
are interior, so the jumps exclude differentiability there. Equality
in \eqref{eq:minimum} would identify a differentiable competitor with
$k_*$ throughout the interval, contradicting either jump.
\end{proof}

\section{A quantitative exponential estimate}

\begin{lemma}\label{lem:quadratic}
For real $m,x,y$ with $m\leq x,y$,
\[
 e^x-e^y-e^y(x-y)\geq\frac{e^m}{2}(x-y)^2.
\]
\end{lemma}
\begin{proof}
The function $t\mapsto e^t-e^mt^2/2$ has nonnegative second derivative
on $[m,\infty)$. Its supporting-line inequality at $y$, applied to $x$,
is exactly the stated bound after rearrangement. This applies in either
order of $x$ and $y$, including the boundary value $m$.
\end{proof}

\section{Quantitative stability}

For a continuous real function $\delta$ on $[0,1]$, define
\[
 \mathcal Q(\delta)=\int_0^1\int_0^s
       2s\bigl(\delta(v)-\delta(s)\bigr)^2\,dv\,ds,
 \qquad \mu=\int_0^1\delta,
 \qquad g=\delta-\mu,
 \qquad G(x)=\int_0^xg.
\]
\begin{lemma}\label{lem:variance}
The actual weighted pair integral satisfies
\begin{equation}\label{eq:variance}
 \mathcal Q(\delta)=\int_0^1(1+x^2)g(x)^2\,dx
                       +2\int_0^1G(x)^2\,dx.
\end{equation}
In particular $\mathcal Q(\delta)\geq\int_0^1g^2$.
\end{lemma}
\begin{proof}
Centering does not change pair differences, and $\int_0^1g=0$.
Let $H(x)=\int_0^xg(v)^2\,dv$. Expanding the inner integral gives
\[
 \mathcal Q(g)=\int_0^1
    \bigl(2xH(x)-4xg(x)G(x)+2x^2g(x)^2\bigr)\,dx.
\]
The function $F(x)=(x^2-1)H(x)-2xG(x)^2$ is continuous up to the
endpoints and differentiable in the interior. Direct differentiation gives
\[
 F'(x)=2xH(x)-4xg(x)G(x)+2x^2g(x)^2
                   -\bigl((1+x^2)g(x)^2+2G(x)^2\bigr).
\]
Since $G(0)=G(1)=H(0)=0$, both endpoint values of $F$ vanish.
The fundamental theorem of calculus proves \eqref{eq:variance}. All
integrands are continuous; no derivative of $\delta$ is required.
The final inequality follows from nonnegativity of the additional terms.
\end{proof}

\begin{theorem}\label{thm:stability}
For $C\geq1$, let $\alpha$ and $k_*$ be as in Theorem~\ref{thm:minimum},
and define
\begin{equation}\label{eq:stabilityconstant}
 c(C)=\left(\frac{12C}{5\alpha}
                +\frac{\log C}{2\alpha^3}\right)^{-1}>0.
\end{equation}
Every $k\in\mathcal A_C$ satisfies
\begin{equation}\label{eq:stability}
 \I(k)-\I(k_*)\geq c(C)\int_0^1|k(v)-k_*(v)|^2\,dv.
\end{equation}
\end{theorem}

\begin{proof}
At $C=1$, both profiles vanish on the unit interval, so the assertion is
immediate. For $C>1$, set $L=\log C$, $\delta=k-k_*$ and
$D=\I(k)-\I(k_*)$. Write the exact deficit as
$D=\mathcal P+\mathcal R$, where
$\mathcal P=\int_0^1 A(v)\delta(v)\,dv\geq0$.
For every pair $v<s$, the differences
$x=k(v)-k(s)$ and $y=k_*(v)-k_*(s)$ are at least $-L$.
Lemma~\ref{lem:quadratic} with $m=-L$ gives
\[
 e^y\bigl(e^{\delta(v)-\delta(s)}-1-\delta(v)+\delta(s)\bigr)
 \geq\frac1{2C}\bigl(\delta(v)-\delta(s)\bigr)^2.
\]
Multiplication by $2s$ and integration yield
$\mathcal R\geq a\mathcal Q(\delta)$, where $a=1/(2C)>0$.
With $\mu,g$ as above and $V=\int_0^1g^2$, Lemma~\ref{lem:variance}
therefore gives
\begin{equation}\label{eq:variancebudget}
 \mathcal P+aV\leq D.
\end{equation}

To control the constant mode, use the complete lower-obstacle weight
\[
 w(v)=\frac{3\max\{\alpha^2-v^2,0\}}{2\alpha^3}
       \quad(0\leq v\leq1).
\]
This is continuous and nonnegative. Polynomial integration gives
\begin{equation}\label{eq:weightmoments}
 \int_0^1w=1,\qquad \int_0^1w^2=\frac6{5\alpha},\qquad
 r:=\int_0^1(w-1)^2=\frac6{5\alpha}-1\geq0.
\end{equation}
On $[0,\alpha]$, the lower obstacle gives $\delta=k\in[0,L]$ and
$A(v)=3(\alpha^2-v^2)$. Elsewhere $w=0$ and $A\delta\geq0$.
Thus the pointwise bound $\delta^2\leq L\delta$ on the support of
$w$ proves
\begin{equation}\label{eq:anchor}
 \int_0^1w\delta^2\leq b\mathcal P,
 \qquad b=\frac{L}{2\alpha^3}\geq0.
\end{equation}

Put $u=\int_0^1w\delta$. Positivity of
$\int_0^1w(\delta-u)^2$ and $\int_0^1w=1$ imply
$u^2\leq\int_0^1w\delta^2\leq b\mathcal P$.
The identity $\int_0^1(w-1)g=u-\mu$ and Cauchy--Schwarz give
$(u-\mu)^2\leq rV$. This instance of Cauchy--Schwarz follows directly
by expanding the nonnegative quadratic integral
$\int_0^1(t(w-1)+g)^2$ for every real $t$.
Consequently
\[
 |\mu|\leq\sqrt{b\mathcal P}+\sqrt{rV},\qquad
 \mu^2\leq (b+r/a)(\mathcal P+aV)\leq(b+r/a)D.
\]
The middle inequality is the two-dimensional Cauchy--Schwarz inequality
applied to $(\sqrt b,\sqrt{r/a})$ and
$(\sqrt{\mathcal P},\sqrt{aV})$.
Finally $\int_0^1\delta^2=V+\mu^2$, and \eqref{eq:variancebudget}
implies $V\leq D/a$. Combining these facts with \eqref{eq:weightmoments}
gives
\[
 \int_0^1\delta^2\leq\left(b+\frac{1+r}{a}\right)D
 =\left(\frac{\log C}{2\alpha^3}+\frac{12C}{5\alpha}\right)D.
\]
The coefficient is positive because $C,\alpha>0$ and $\log C\geq0$.
This proves \eqref{eq:stability}. No optimality of $c(C)$ is asserted.
\end{proof}

\section{Two hemispheres and near-equatorial symmetry}

\begin{corollary}\label{cor:hemispheres}
Let $h:\R\to\R$ be continuous on $[-1,1]$ and satisfy
$0\leq h\leq\log C$ there, where $C\geq1$. Define
\[
 k_+(v)=h(v),\qquad k_-(v)=h(-v),\qquad
 \mathcal D=\I(k_+)+\I(k_-)-2\I(k_*).
\]
Then
\begin{align}
 \int_{-1}^1|h(v)-k_*(|v|)|^2\,dv
       &\leq\frac{\mathcal D}{c(C)},\label{eq:fullprofile}\\
 \int_{-1}^1|h(v)-h(-v)|^2\,dv
       &\leq\frac{4\mathcal D}{c(C)},\label{eq:reflection}\\
 \int_{-1}^1\left|h(v)-\frac{h(v)+h(-v)}2\right|^2\,dv
       &\leq\frac{\mathcal D}{c(C)}.\label{eq:symmetrization}
\end{align}
No reflection or balance condition is imposed on $h$.
\end{corollary}
\begin{proof}
Apply Theorem~\ref{thm:stability} to each hemisphere. Splitting the
integral at zero and substituting $v\mapsto-v$ on the southern half
identifies their summed squared errors with the left side of
\eqref{eq:fullprofile}. At each northern point, the elementary inequality
$(x-y)^2\leq2x^2+2y^2$, applied to the two errors against $k_*$,
bounds the reflection error. That error has equal integrals on the two
halves, giving the factor four in \eqref{eq:reflection}.
The symmetrization error is pointwise one quarter of the squared
reflection error, proving \eqref{eq:symmetrization}.
At cap one all three errors vanish directly.
\end{proof}

\section{Smooth balanced profiles}

Let $a:\R\to\R$ be globally smooth, positive on $[-1,1]$, and balanced:
\begin{equation}\label{eq:balance}
 \int_{-1}^1 v a(v)\,dv=0.
\end{equation}
Define the actual moment coordinate and warping function by
\begin{equation}\label{eq:warping}
 x(v)=\int_0^v a(\xi)\,d\xi,
 \qquad f(v)=2\int_v^1 \xi a(\xi)\,d\xi.
\end{equation}

\begin{lemma}\label{lem:pole-factors}
The functions $x$ and $f$ are globally smooth, with $x'=a$ and
$f'=-2va$. The coordinate $x$ is strictly increasing on $[-1,1]$.
Moreover, $f(-1)=f(1)=0$, $f>0$ on $(-1,1)$, and there exist globally
smooth real functions $b,c$ such that, on the whole real line,
\begin{equation}\label{eq:pole-factors}
 f(v)=(1-v^2)b(v),\qquad a(v)^2-b(v)^2=(1-v^2)c(v).
\end{equation}
Here $b>0$ on $[-1,1]$ and $b(\pm1)=a(\pm1)$.
For a constant profile $a=r>0$, one may take $b=r$ and $c=0$.
\end{lemma}

\begin{proof}
The derivative formulas follow from the fundamental theorem of calculus;
they also imply global smoothness. Positivity of $a$ makes the integral
of $a$ between any two distinct points of $[-1,1]$ positive, proving strict
monotonicity of $x$. The northern endpoint value is immediate and the
southern one is exactly the balance condition. For $0\leq v<1$, the
integrand in the northern expression for $f$ is positive on $(v,1)$.
For $-1<v<0$, balance rewrites that expression as
$f(v)=-2\int_{-1}^v \xi a(\xi)\,d\xi>0$.

For a globally smooth function $p$ with $p(t)=0$, the fundamental theorem
gives the smooth factorization
\[
 p(v)=(v-t)\int_0^1 p'(t+s(v-t))\,ds.
\]
Applying it successively at the two distinct roots $-1,1$ gives a globally
smooth factor of $1-v^2$. Apply this first to $f$ to obtain $b$.
Differentiating $f=(1-v^2)b$ at each pole and using $f'=-2va$ yields
$b(\pm1)=a(\pm1)$. Interior positivity follows from $f>0$ and $1-v^2>0$;
the endpoint values give positivity at the poles. Thus $a^2-b^2$ vanishes
at both poles, and the same factorization gives $c$.
Finally, for $a=r$ direct integration gives $f(v)=r(1-v^2)$, which verifies
the stated constant-profile factors.
\end{proof}

\begin{theorem}\label{thm:smooth-metric}
Let $a$ satisfy the hypotheses of Lemma~\ref{lem:pole-factors}, and let
$b,c$ be the factors supplied there. On the standard unit sphere
$S^2\subset\R^3$, let $z$ be the third coordinate and $g_0$ the round metric.
Then
\begin{equation}\label{eq:global-metric}
 g=b(z)g_0+\frac{c(z)}{b(z)}\,dz\otimes dz
\end{equation}
is a smooth positive Riemannian metric on the whole sphere, including both
poles. For the constant profile $a=r>0$ with $b=r,c=0$, it is exactly $r g_0$.
\end{theorem}

\begin{proof}
The height takes values in $[-1,1]$, where $b$ is positive, so both scalar
coefficients in \eqref{eq:global-metric} are smooth on the sphere. The
one-form $dz$ is the actual differential of the smooth coordinate function,
hence the displayed tensor is smooth at the poles as well as elsewhere.
For $p\in S^2$ and $v\in T_pS^2$, orthogonality $p\cdot v=0$ gives
\[
 dz_p(v)=\bigl(e_3-z(p)p\bigr)\cdot v,\qquad
 |dz_p(v)|^2\leq\bigl(1-z(p)^2\bigr)g_0(v,v).
\]
Write $q=c(z)/b(z)$. If $q\geq0$, positivity follows from $b(z)>0$.
If $q<0$, the same bound and the second factorization in
\eqref{eq:pole-factors} give
\[
 g(v,v)\geq\bigl[b(z)+q(1-z^2)\bigr]g_0(v,v)
       =\frac{a(z)^2}{b(z)}g_0(v,v)>0\qquad(v\ne0).
\]
The constant-profile formula follows directly from $b=r,c=0$.
\end{proof}

\section{Cylinder coordinates and area}

For $-1<v<1$ and $\theta\in\R$, put $\rho(v)=\sqrt{1-v^2}$ and
\[
 \Phi(v,\theta)=(\rho(v)\cos\theta,\rho(v)\sin\theta,v)\in S^2.
\]
The angle is a real parameter, so no global injectivity is asserted.

\begin{lemma}\label{lem:cylinder-metric}
The map $\Phi$ is smooth and has injective differential. Its height is $v$,
and the metric of Theorem~\ref{thm:smooth-metric} satisfies
\begin{equation}\label{eq:cylinder-metric}
 \Phi^*g=\frac{a(v)^2}{f(v)}\,dv^2+f(v)\,d\theta^2.
\end{equation}
\end{lemma}

\begin{proof}
The identity $\rho^2+v^2=1$ gives the sphere constraint, and $\rho>0$
gives smoothness on the stated domain. The actual derivatives are
\[
 \Phi_v=\left(-\frac v\rho\cos\theta,-\frac v\rho\sin\theta,1\right),
 \qquad
 \Phi_\theta=(-\rho\sin\theta,\rho\cos\theta,0).
\]
Their round inner products are $\rho^{-2}$, $0$, and $\rho^2$,
respectively. In particular the differential is injective. Since
$dz(\Phi_v)=1$ and $dz(\Phi_\theta)=0$, the radial coefficient of the
pullback of \eqref{eq:global-metric} is
\[
 \frac b{1-v^2}+\frac cb
 =\frac{b^2+(1-v^2)c}{(1-v^2)b}
 =\frac{a^2}{f}.
\]
The angular coefficient is $(1-v^2)b=f$, and the mixed coefficient
vanishes. These equalities give \eqref{eq:cylinder-metric}.
\end{proof}

\begin{theorem}\label{thm:volume-height}
Write $d\mu_0$ for round area measure. The metric of
Theorem~\ref{thm:smooth-metric} has the global area identity
\begin{equation}\label{eq:volume-profile}
 d\mu_g=a(z)\,d\mu_0.
\end{equation}
For every $F:\R\to\R$ continuous on $[-1,1]$,
\begin{equation}\label{eq:height-integral}
 \int_{S^2}F(z)\,d\mu_g
   =2\pi\int_{-1}^1 a(v)F(v)\,dv.
\end{equation}
\end{theorem}

\begin{proof}
The round gradient of height at $p$ is $e_3-z(p)p$, of squared length
$1-z(p)^2$. In a round orthonormal tangent basis, the determinant of
$b g_0+q\,dz\otimes dz$ is
\[
 b^2+bq\,|dz|_{g_0}^2=b^2+c(1-z^2)=a^2,
 \qquad q=c/b.
\]
Thus its area density relative to $g_0$ is $a>0$. The computation is
valid at the poles and in every chart, proving the measure identity
\eqref{eq:volume-profile}.

For completeness, the round height disintegration follows directly
from the two hemisphere graphs. On the open unit disk in $\R^2$,
the upper graph $y\mapsto(y,\sqrt{1-|y|^2})$ has area density
$(1-|y|^2)^{-1/2}$. The lower graph has the same density and negative
height. The equator has round area zero. Radial integration and the
substitution $u=\sqrt{1-r^2}$ therefore give, for continuous $G$ on
$[-1,1]$,
\[
 \int_{S^2}G(z)\,d\mu_0
 =2\pi\int_0^1
   \frac{r\,[G(\sqrt{1-r^2})+G(-\sqrt{1-r^2})]}{\sqrt{1-r^2}}\,dr
 =2\pi\int_{-1}^1G(u)\,du.
\]
All integrals converge: $G$ is bounded and
$r/\sqrt{1-r^2}$ is integrable on $(0,1)$.
Applying this formula to $G=aF$ and using \eqref{eq:volume-profile}
proves \eqref{eq:height-integral}.
\end{proof}

\begin{theorem}\label{thm:curvature-profile}
The Gauss curvature of the metric in Theorem~\ref{thm:smooth-metric} is
\begin{equation}\label{eq:curvature-profile}
 K_g(p)=\frac1{a(z(p))}\qquad(p\in S^2).
\end{equation}
Equivalently, its scalar curvature is $2/a(z)$, and every nondegenerate
tangent two-plane has sectional curvature $1/a(z)$.
\end{theorem}

\begin{proof}
For any point with $-1<z<1$, choosing the argument of its first two
coordinates gives a preimage under $\Phi$. The differential computed in
Lemma~\ref{lem:cylinder-metric} is invertible between the two tangent spaces,
so the inverse function theorem makes $\Phi$ a local coordinate map there.
Set $E=a^2/f$ and $F=f$. The nonzero Levi--Civita coefficients of
$E\,dv^2+F\,d\theta^2$ are
\[
 \Gamma^v_{vv}=\frac{E'}{2E},\qquad
 \Gamma^v_{\theta\theta}=-\frac{F'}{2E},\qquad
 \Gamma^\theta_{v\theta}=\Gamma^\theta_{\theta v}=\frac{F'}{2F}.
\]
Substitution in the curvature tensor gives
\[
 \operatorname{Ric}_{vv}
 =-\frac{F''}{2F}+\frac{(F')^2}{4F^2}+\frac{E'F'}{4EF}.
\]
In dimension two, contracting the curvature tensor gives
$\operatorname{Ric}=K_g g$ and scalar curvature $2K_g$. Consequently
\[
 K_g
 =-\frac{F''}{2EF}+\frac{(F')^2}{4EF^2}+\frac{E'F'}{4E^2F}
 =-\frac1{2\sqrt{EF}}\left(\frac{F'}{\sqrt{EF}}\right)'.
\]
Here $\sqrt{EF}=a>0$ and $F'=f'=-2va$, so the last expression is $1/a$.
This computes the intrinsic curvature of the same metric through its
actual local pullback.

The cylinder image is precisely the complement of the two poles and is
dense in $S^2$. Both the scalar curvature of the smooth metric and
$2/a(z)$ are continuous on the whole sphere. Their equality therefore
extends over the poles. The two-dimensional scalar/sectional curvature
identity proves the equivalent assertions.
\end{proof}

\section{Smooth one-forms and the complete dissipation}

Retain the metric constructed from $a,b,c$, and write $D=(a,b,c)$ for these data.
Let $x,y,z$ be the ambient coordinate functions restricted to the unit sphere. Put
$\kappa=-y\,dx+x\,dy$.

\begin{lemma}\label{lem:global-forms}
For globally smooth $r,s:\R\to\R$, the formulas
\[
 M_D(r)=a(z)r(z)\,dz,\qquad Z_D(s)=b(z)s(z)\,\kappa
\]
define smooth one-forms on the entire sphere. Both vanish at the poles,
and their pullbacks by the actual cylinder map are
\begin{equation}\label{eq:forms-cylinder}
 \Phi^*M_D(r)=a(v)r(v)\,dv,\qquad
 \Phi^*Z_D(s)=f(v)s(v)\,d\theta.
\end{equation}
The constant unit probes $M_D(1)$ and $Z_D(1)$ are nonzero.
\end{lemma}

\begin{proof}
Every factor is globally smooth on the sphere: the one-forms $dx,dy,dz$
are the actual differentials of its smooth coordinate functions. At a pole
$x=y=0$, hence $\kappa=0$; also $dz=0$ on the tangent plane there. This proves
the pole statements directly from the global definitions.
The derivative formulas in Lemma~\ref{lem:cylinder-metric} give
$\Phi^*dz=dv$ and $\Phi^*\kappa=(1-v^2)d\theta$.
Using $f=(1-v^2)b$ proves \eqref{eq:forms-cylinder}.
At $v=0$, evaluation on the respective cylinder coordinate directions
gives $a(0)>0$ and $f(0)>0$ for the unit probes, proving nonvanishing.
\end{proof}

\begin{lemma}\label{lem:canonical-derivatives}
For a smooth Riemannian metric $g$ and a smooth covariant tensor $T$ on a
smooth Hausdorff manifold, its first and second total covariant derivatives
are smooth and uniquely determined by their covariant-derivative realization
laws. Both assignments are real-linear in $T$.
For a one-form $h$, writing $N=\nabla h$ and $H=\nabla N$, these laws imply
\[
 N(X,Y)=X(h(Y))-h(\nabla_XY),
\]
\[
 H(X,Y,Z)=X(N(Y,Z))-N(\nabla_XY,Z)-N(Y,\nabla_XZ)
\]
for smooth vector fields $X,Y,Z$.
\end{lemma}

\begin{proof}
Use the Levi--Civita connection of the given metric. In local coordinates
the covariant derivative of $T$ is its ordinary derivative minus one
Christoffel term for each covariant slot. These expressions are smooth
and transform tensorially, producing the first total derivative as a
global smooth tensor. Repeating the construction produces the second.
The displayed formulas are its one-form and two-tensor product rules.
To prove uniqueness, extend any prescribed tangent vector at a point to
a global smooth vector field using a coordinate neighborhood and a smooth
cutoff. The realization laws then determine every component in a tangent
basis at that point. Applying the same argument first to $N$ and then to
$H$ gives uniqueness of the pair. Real-linearity follows from the
connection's linearity in its tensor argument and iteration.
\end{proof}

On a compact smooth Riemannian surface $(\Sigma,g)$ without boundary, write
$K_g=\operatorname{Scal}_g/2$ and use the trace convention
\[
 \Delta^\nabla h=\operatorname{tr}_{g,12}(\nabla^2h),\qquad
 A_g h=\frac12\bigl(\nabla h+(\nabla h)^{\mathsf T}\bigr)
       -\frac12\operatorname{tr}_g(\nabla h)\,g.
\]
Here the first two slots of the total second derivative are traced.
Thus $A_g h$ is the actual trace-free symmetric first derivative.
Define the normalized complete dissipation by
\begin{equation}\label{eq:complete-dissipation}
 Q_g(h)=\int_\Sigma
 \left(
 |\Delta^\nabla h|_g^2+K_g^2|h|_g^2
 -K_g|\nabla h|_g^2+2K_g|A_g h|_g^2
 \right)\,d\mu_g .
\end{equation}

\begin{theorem}\label{thm:dissipation-quadratic}
For every smooth one-form, the density in \eqref{eq:complete-dissipation}
is smooth and integrable. Its polarization
\[
 B_g(h,k)=\int_\Sigma
 \left(
 \langle\Delta^\nabla h,\Delta^\nabla k\rangle_g
 +K_g^2\langle h,k\rangle_g
 -K_g\langle\nabla h,\nabla k\rangle_g
 +2K_g\langle A_g h,A_g k\rangle_g
 \right)\,d\mu_g
\]
is a well-defined symmetric real bilinear form, with $Q_g(h)=B_g(h,h)$.
In particular, for $c\in\R$,
\[
 Q_g(ch)=c^2Q_g(h),\qquad
 Q_g(h\pm k)=Q_g(h)\pm2B_g(h,k)+Q_g(k),
\]
and
\[
 Q_g(h+k)+Q_g(h-k)=2Q_g(h)+2Q_g(k).
\]
\end{theorem}

\begin{proof}
Lemma~\ref{lem:canonical-derivatives} supplies the actual smooth first and
second derivatives. Metric contraction and permutation of covariant slots
preserve smoothness. The symmetric part has the same metric trace as the
original tensor; subtracting half this trace times $g$ gives zero trace in
dimension two. This proves the asserted smooth trace-free symmetric
construction and its real-linearity.

The curvature and all metric inner products of smooth tensor fields are
smooth. Hence each displayed density is smooth. Compactness and finiteness
of Riemannian volume make these densities integrable. The derivative,
rough-trace and trace-free symmetric operators are real-linear.
The metric inner product is symmetric bilinear, so each of the four
terms in $B_g$ is symmetric bilinear in $(h,k)$. Their integrability
justifies integration of the linearity identities. Evaluation at $h=k$
gives \eqref{eq:complete-dissipation}; expansion and scalar multiplication
then give the stated quadratic identities.
\end{proof}

\begin{corollary}\label{cor:profile-dissipation}
For the produced rotational metric and every smooth one-form $h$,
\[
 Q_g(h)=\int_{S^2}a(z)
 \left(
 |\Delta^\nabla h|_g^2+\frac{|h|_g^2}{a(z)^2}
 -\frac{|\nabla h|_g^2}{a(z)}
 +\frac{2|A_g h|_g^2}{a(z)}
 \right)\,d\mu_0.
\]
The displayed round-measure integrand is integrable. All operators and
norms in this formula belong to the same metric $g$.
\end{corollary}

\begin{proof}
Substitute $K_g=1/a(z)$ from Theorem~\ref{thm:curvature-profile} and
$d\mu_g=a(z)\,d\mu_0$ from Theorem~\ref{thm:volume-height}.
The weighted density is continuous on the compact sphere, which proves
its integrability against round area.
\end{proof}

\section{The exact zonal action}

Continue with a globally smooth positive balanced profile $a$ and the same
produced metric $g$. Write $f(v)=2\int_v^1\xi a(\xi)\,d\xi$ and
$H(v)=f(v)/a(v)$. Since $a$ is positive on the compact interval,
$H$ is smooth on a neighborhood of $[-1,1]$. For a globally smooth
real probe $r$, set
\begin{align}
 L_a r(v)&=H(v)r''(v)+(H'(v)-2v)r'(v)-r(v),\label{eq:meridional-operator}\\
 J_a[r]&=\int_{-1}^1\left\{
 H(L_a r)^2+(H-2v^2)r^2+2vHrr'\right\}\,dv.
 \label{eq:meridional-action}
\end{align}
These are ordinary integrals of continuous functions on the closed interval.

\begin{lemma}\label{lem:meridional-derivatives}
Let $U(v)=\int_0^v a(t)r(t)\,dt$. The global meridional form is
$M(r)=d(U\circ z)$, and its canonical derivatives satisfy
\begin{align}
 \nabla M(r)&=a(z)r'(z)\,dz\otimes dz-zr(z)g,
 \label{eq:meridional-first}\\
 \Delta^\nabla M(r)&=(L_a r)(z)\,dz.
 \label{eq:meridional-rough}
\end{align}
In particular, everywhere on the sphere,
\begin{align*}
 |dz|_g^2&=f(z)/a(z)^2, & |M(r)|_g^2&=f(z)r(z)^2,\\
 \operatorname{tr}_g\nabla M(r)&=H(z)r'(z)-2zr(z),\\
 |\nabla M(r)|_g^2&=(H(z)r'(z)-zr(z))^2+(zr(z))^2,\\
 |A_gM(r)|_g^2&=\tfrac12(H(z)r'(z))^2.
\end{align*}
No equatorial symmetry of $a$ or $r$ is assumed.
\end{lemma}

\begin{proof}
The fundamental theorem of calculus and the chain rule give
$d(U\circ z)=a(z)r(z)dz$, including at the poles. On the genuine open
cylinder of Lemma~\ref{lem:cylinder-metric}, the metric is
$E\,dv^2+f\,d\theta^2$, where $E=a^2/f$ and $f'=-2va$.
The Levi--Civita formula therefore gives
\[
 \Gamma^v_{vv}=\frac{a'}a+\frac{va}{f},\qquad
 \Gamma^v_{\theta\theta}=\frac{vf}{a},\qquad
 \Gamma^v_{v\theta}=0.
\]
The covariant derivative of the actual covector $ar\,dv$ has components
$ar'-va^2r/f$, $-vfr$, and zero mixed components. This is exactly
\eqref{eq:meridional-first}. Both sides are smooth global tensors;
the cylinder has dense image, so the identity holds at the poles as well.

The round height gradient has squared length $1-z^2$. The metric tensor
and the pole-factor identity show that the $g$-dual of $dz$ is
$(b/a^2)\operatorname{grad}_{g_0}z$. Thus $|dz|_g^2=f/a^2$ globally.
Contracting \eqref{eq:meridional-first} gives the displayed trace and norm
formulas. Equivalently, away from the poles its two eigenvalues are
$Hr'-vr$ and $-vr$; subtracting half the trace leaves eigenvalues
$Hr'/2$ and $-Hr'/2$. Continuity again includes both poles.

For completeness, the exact-form commutation formula with the convention
$\Delta^\nabla=\operatorname{tr}_g\nabla^2$ is
\[
 \Delta^\nabla dF=d(\Delta_g F)+\operatorname{Ric}_g(\nabla F,\cdot).
\]
It follows by commuting the first two covariant derivatives of $dF$
and contracting their curvature term. All derivatives here are those
of the same Levi--Civita connection. On a surface
$\operatorname{Ric}_g=K_g g$, and Theorem~\ref{thm:curvature-profile}
gives $K_g=1/a(z)$. Since
$\Delta_g(U\circ z)=(Hr'-2vr)(z)$, differentiating this scalar identity
gives
\[
 \Delta^\nabla M(r)
 =\bigl(Hr''+(H'-2v)r'-2r+r\bigr)(z)\,dz.
\]
This is \eqref{eq:meridional-rough}.
\end{proof}

\begin{theorem}\label{thm:meridional-reduction}
For every globally smooth real probe $r$, the complete action of the
original global meridional section is
\[
 Q_g(M(r))=2\pi J_a[r].
\]
For every constant positive profile $a\equiv c$, the fixed unit probe
satisfies $J_a[1]=4/3$ and $Q_g(M(1))=8\pi/3$.
\end{theorem}

\begin{proof}
Insert Lemma~\ref{lem:meridional-derivatives} into the four terms of
\eqref{eq:complete-dissipation} and multiply the density by $a$.
The rough term becomes $H(L_a r)^2$, and the curvature-square term
becomes $Hr^2$. The two first-derivative terms together are
\[
 -\bigl((Hr'-vr)^2+(vr)^2\bigr)+(Hr')^2
 =2vHrr'-2v^2r^2.
\]
Their sum is precisely the integrand in \eqref{eq:meridional-action}.
The volume and height identities in Theorem~\ref{thm:volume-height}
then give the factor $2\pi$. This calculation retains all four terms
and requires no endpoint integration by parts. The continuous interval
density and the smooth sphere density are genuinely integrable.

If $a\equiv c>0$, then $f=c(1-v^2)$ and $H=1-v^2$.
For $r\equiv1$, $L_a r=-1$ and the density is $2-4v^2$.
Its integral is $4/3$, giving the stated geometric normalization for
the same metric and section at every positive scale $c$.
\end{proof}

\begin{lemma}\label{lem:pullback-dissipation}
Let $g$ be a smooth metric on a compact smooth surface and $\phi$ a smooth
self-diffeomorphism. Pull back a smooth covariant tensor by its actual
derivative in every slot. Then
\[
 \nabla^{\phi^*g}(\phi^*T)=\phi^*(\nabla^g T),\qquad
 (\nabla^{\phi^*g})^2(\phi^*T)=\phi^*((\nabla^g)^2T).
\]
The rough Laplacian, tensor norms and trace-free symmetric projection
transform under the same pullback. In particular,
\[
 Q_{\phi^*g}(\phi^*h)=Q_g(h),\qquad
 B_{\phi^*g}(\phi^*h,\phi^*k)=B_g(h,k).
\]
These statements require no orientation-preserving hypothesis.
\end{lemma}

\begin{proof}
The pulled-back connection is metric compatible and torsion free for
$\phi^*g$, so uniqueness identifies it with that metric's Levi--Civita
connection. Evaluating a covariant derivative on smooth vector fields,
the scalar chain rule and the connection identity transform its derivative
term and every slot correction. This proves the first identity; apply it
again for the second. Contraction by the inverse pulled-back metric
commutes with pullback, proving the rough-trace, norm and trace-free
symmetric identities. Curvature and its scalar contraction transform
through this same connection. Thus each of the four density terms pulls
back by composition with $\phi$. Riemannian volume transforms with the
absolute Jacobian, so change of variables proves the action identity
for both orientations. Polarization proves the pairing identity.
The tensor pullback itself satisfies the identity and contravariant
composition laws by the ordinary derivative chain rule, hence also both
inverse-cancellation laws.
\end{proof}

\begin{lemma}\label{lem:zonal-orthogonality}
For the produced rotational metric and globally smooth probes $r,s$,
\[
 B_g(M(r),Z(s))=0,\qquad
 Q_g(M(r)+Z(s))=Q_g(M(r))+Q_g(Z(s)).
\]
\end{lemma}

\begin{proof}
The actual sphere reflection $\rho(x,y,z)=(x,-y,z)$ preserves the round
metric, height and its differential. It therefore preserves the produced
metric tensor as well. Its covector pullback fixes $dz$ and sends
$\kappa=-y\,dx+x\,dy$ to $-\kappa$. Consequently $\rho^*M(r)=M(r)$
and $\rho^*Z(s)=-Z(s)$, with the derivative of $\rho$ included in both
pullbacks. Lemma~\ref{lem:pullback-dissipation} and bilinearity give
$B_g(M(r),Z(s))=-B_g(M(r),Z(s))$, hence the pairing vanishes.
The quadratic expansion then gives the sum identity. Reflection fixes
height rather than reversing it, so no equatorial symmetry is needed.
\end{proof}

\begin{lemma}\label{lem:parallel-area}
For the same produced metric, define the smooth tensor
\[
 \Omega_x(V,W)=a(z(x))\det(x,d\iota_xV,d\iota_xW),
\]
where $\iota:S^2\hookrightarrow\mathbb R^3$ is the actual inclusion.
It is alternating, has squared tensor norm $2$, and is parallel for
the metric's Levi--Civita connection.
\end{lemma}

\begin{proof}
The determinant is a finite sum of products of ambient coordinate
functions and their differentials, so the tensor is smooth globally.
For tangent vectors $V,W$, the square of the round determinant form is
$g_0(V,V)g_0(W,W)-g_0(V,W)^2$: expand the scalar triple product using
$|x|=1$ and $x\perp d\iota V,d\iota W$. The metric determinant identity
from Theorem~\ref{thm:volume-height} multiplies this Gram determinant
by $a(z)^2$. Therefore $\Omega(e_1,e_2)^2=1$ in every $g$-orthonormal
basis. Alternation then gives $|\Omega|_g^2=2$.

Covariant differentiation preserves alternation. On a two-dimensional
tangent space, every alternating two-tensor is a multiple of the
nonzero tensor $\Omega$. Metric compatibility and differentiation of
its constant squared norm give
$\langle\nabla_X\Omega,\Omega\rangle_g=0$ for every tangent vector $X$.
That multiple must vanish, proving $\nabla\Omega=0$.
\end{proof}

\begin{theorem}\label{thm:zonal-reduction}
For the produced metric and its area form from Lemma~\ref{lem:parallel-area},
define the actual smooth Hodge rotation by
\[
 (\mathcal Rh)_x(Y)=\Omega_x(h_x^\sharp,Y).
\]
It acts on the final covector slot of both canonical covariant derivatives
and commutes with the rough Laplacian. It preserves the complete density
and action of every smooth one-form. With the outward determinant convention,
\[
 \mathcal R M(r)=-Z(r),\qquad Q_g(Z(r))=2\pi J_a[r].
\]
Consequently, for every pair of globally smooth real probes $r,s$,
\begin{equation}\label{eq:full-zonal-reduction}
 Q_g(M(r)+Z(s))=2\pi\bigl(J_a[r]+J_a[s]\bigr).
\end{equation}
\end{theorem}

\begin{proof}
The metric sharp map and contraction of smooth tensors produce a smooth
global one-form. Metric compatibility and $\nabla\Omega=0$ give
\[
 (\nabla\mathcal Rh)(X,Y)=\Omega((\nabla_Xh)^\sharp,Y).
\]
To differentiate once more, extend $Y$ to a smooth vector field and
differentiate the first-slot contraction of $\nabla h$. The derivative
of that vector field occurs on both sides; subtracting the connection
correction in the second derivative cancels it. The correction in the
final covector slot is already part of the first-derivative formula.
Thus
\[
 (\nabla^2\mathcal Rh)(X,Y,Z)
   =\Omega((\nabla^2h)(X,Y,\cdot)^\sharp,Z).
\]
Taking the actual metric trace in $X,Y$ proves
$\Delta^\nabla\mathcal Rh=\mathcal R\Delta^\nabla h$.

In an orthonormal frame put $\varepsilon=\Omega(e_1,e_2)$, so
$\varepsilon^2=1$. The covector components $(u,v)$ rotate to
$(-\varepsilon v,\varepsilon u)$. This preserves their norm and the
norm of every row of the first derivative. If a two-tensor has matrix
$\left(\begin{smallmatrix}A&B\\C&D\end{smallmatrix}\right)$, its
trace-free symmetric part has squared norm
$((A-D)^2+(B+C)^2)/2$. Rotating its final slot replaces its rows by
$(-\varepsilon B,\varepsilon A)$ and
$(-\varepsilon D,\varepsilon C)$, which preserves this expression.
The rough, zeroth-order, first-order and Ahlfors norms in the complete
density therefore all agree before and after rotation. Integrating
proves $Q_g(\mathcal Rh)=Q_g(h)$ without a curvature-sign assumption.

For the actual sphere, the included round height gradient is
$e_3-zx$. Direct determinant expansion gives
$\Omega_0(\operatorname{grad}_{g_0}z,Y)=-\kappa(Y)$.
The metric-dual formula used in Lemma~\ref{lem:meridional-derivatives}
gives $M(r)^\sharp=(b/a)r\operatorname{grad}_{g_0}z$.
Since $\Omega=a\Omega_0$, it follows globally that
$\mathcal RM(r)=-br\kappa=-Z(r)$. The quadratic action is unchanged
by a minus sign, so the azimuthal action equals the meridional action.
Combine Theorem~\ref{thm:meridional-reduction} with
Lemma~\ref{lem:zonal-orthogonality} to obtain
\eqref{eq:full-zonal-reduction}.
\end{proof}

\section{The smooth Haar projection and complete action split}

\begin{lemma}\label{lem:circle-action}
For a unit complex number $\zeta$, the ambient map
\[
 (x,y,z)\longmapsto
 \bigl((\operatorname{Re}\zeta)x-(\operatorname{Im}\zeta)y,
       (\operatorname{Im}\zeta)x+(\operatorname{Re}\zeta)y,z\bigr)
\]
restricts to a smooth diffeomorphism of the actual unit sphere and preserves
every produced metric $g$. Write $R_\theta$ for $\zeta=e^{i\theta}$.
Then $R_0=\mathrm{id}$, $R_{\theta+\phi}=R_\theta\circ R_\phi$,
$R_{-\theta}=R_\theta^{-1}$, and $R_{\theta+2\pi}=R_\theta$.
The map $(\theta,x)\mapsto R_\theta x$ is jointly smooth, including at
the poles, and its derivative in a sphere tangent direction is the
restriction of the same ambient orthogonal map.
\end{lemma}

\begin{proof}
The identity $(\operatorname{Re}\zeta)^2+(\operatorname{Im}\zeta)^2=1$
proves orthogonality; complex multiplication gives the group law.
Restriction to the sphere preserves the literal map and its derivative.
It fixes height and its differential and preserves the round metric,
so it preserves the produced tensor
$b(z)g_0+(c(z)/b(z))dz\otimes dz$.
The real-angle formula consists of sines, cosines and ambient linear
coordinates, hence is jointly smooth before restriction. The sphere
restriction remains smooth everywhere and has no pole singularity.
\end{proof}

\begin{lemma}\label{lem:tensor-interval-integral}
Let $g$ be a fixed smooth Riemannian metric on a finite-dimensional smooth
manifold without boundary. A jointly smooth real-parameter family $T_t$
of smooth covariant tensors has a smooth fiberwise integral over every
finite oriented interval $[a,b]$. Evaluation and the actual covariant
derivative commute with that integral:
\[
 \left(\int_a^b T_t\,dt\right)_x(V_1,\ldots,V_s)
 =\int_a^b(T_t)_x(V_1,\ldots,V_s)\,dt,
 \qquad
 \nabla\int_a^bT_t\,dt=\int_a^b\nabla T_t\,dt.
\]
Pairing with a fixed tensor, metric traces and the trace-free symmetric
projection also commute with this integral.
\end{lemma}

\begin{proof}
Each fiber is finite dimensional, so its continuous parameter curve
has a genuine Bochner integral on a compact interval. Fix a bundle
trivialization near a base point. Its fiber map is continuous linear
and therefore commutes with integration. In these fixed coordinates,
smooth parameter integration proves smoothness of the integral section.
For each derivative order, compactness of the parameter interval and
joint regularity provide the local bounds needed to differentiate under
the integral; no single bound for all orders is required.

Evaluate the covariant derivative on smooth vector fields. Its scalar
derivative term passes through the integral by the parameter derivative
formula. Each connection correction is evaluation of $T_t$ on fixed
tangent vectors at the point under consideration, so evaluation and the
finite sum of corrections pass through the same integral. This proves
the claimed identity for the genuine covariant derivative. Finally,
metric pairing and each stated contraction are continuous linear maps
on the fixed tensor fiber, which proves their integral identities.
\end{proof}

For a covariant tensor $T$, the pullback $R_\theta^*T$ includes
$dR_\theta$ in every covariant slot. This family is jointly smooth:
the tangent map of the joint action on a vector with zero angle component
is exactly the fixed-angle derivative, and smooth tensor evaluation
preserves joint smoothness. We may consequently define $P$ on all smooth
covariant tensors by normalized fiberwise integration. For a one-form its
literal evaluation is
\begin{equation}\label{eq:haar-covector-average}
 (Ph)_x(V)=\frac1{2\pi}\int_0^{2\pi}
 h_{R_\theta x}((dR_\theta)_xV)\,d\theta.
\end{equation}

\begin{theorem}\label{thm:haar-projection}
The operator $P$ produces smooth global tensors, is real-linear and
idempotent, and satisfies $R_\phi^*PT=PT$ for every circle rotation.
Moreover $PT=T$ if and only if $T$ is invariant under that actual circle
action. In particular $PM(r)=M(r)$ for every globally smooth probe;
the meridional unit form is a nonzero element of the image for every
produced metric.
\end{theorem}

\begin{proof}
Smoothness and evaluation follow from Lemma~\ref{lem:tensor-interval-integral}.
The tensor integrals are genuinely integrable, so linearity follows from
ordinary integral linearity. The group law changes the integrand of
$P(R_\phi^*T)$ or $R_\phi^*PT$ by an angular translation. Every evaluated
orbit is $2\pi$-periodic; its integral over one period is unchanged by
that translation. Thus both expressions equal $PT$.
An invariant tensor has a constant orbit, whose normalized integral
equals the original tensor. Apply this fact to the already invariant
$PT$ to obtain $P^2T=PT$ and the stated characterization of its range.
Every unit complex number is $e^{i\phi}$ for some real $\phi$, so these
identities apply to the whole circle group. The actual rotations preserve
$z$ and $dz$, hence fix $a(z)r(z)dz$. Its unit-probe member is nonzero
by Lemma~\ref{lem:global-forms}.
\end{proof}

\begin{theorem}\label{thm:haar-derivatives}
For the same produced metric, the actual first and second covariant
derivatives commute with $P$ in every tensor rank. For smooth one-forms,
\[
 \Delta^\nabla Ph=P\Delta^\nabla h,\qquad A_g(Ph)=P(A_gh).
\]
All these statements are global and include the poles.
\end{theorem}

\begin{proof}
Every $R_\theta$ preserves $g$. Lemma~\ref{lem:pullback-dissipation}
therefore gives $\nabla R_\theta^*T=R_\theta^*\nabla T$ for the same
connection. Pass this identity through the actual integral using
Lemma~\ref{lem:tensor-interval-integral}. Apply the result again to
$\nabla T$ for the second derivative. Metric trace commutes with this
integral and with isometric pullback, giving the rough-Laplacian identity.
The same argument for the trace-free symmetric projection, after the
first derivative, proves the Ahlfors identity.
\end{proof}

\begin{theorem}\label{thm:haar-split}
For every produced metric and smooth one-forms $h,k$, the actual Haar
projection is self-adjoint for the complete four-term polarization:
\[
 B_g(Ph,k)=B_g(h,Pk),\qquad B_g(Ph,k-Pk)=0.
\]
The original complete action has the exact decomposition
\begin{equation}\label{eq:haar-action-split}
 Q_g(h)=Q_g(Ph)+Q_g(h-Ph).
\end{equation}
No curvature pinching assumption is required for these identities.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:haar-derivatives} and the linear contraction identities
apply to each term of the original polarized density: the rough pairing,
the $K^2$-weighted form pairing, the negative $K$-weighted derivative
pairing, and the positive $2K$-weighted Ahlfors pairing. The resulting
rotated density is jointly continuous on the angle interval times the
compact sphere. Its product integral is finite, so Fubini gives exactly
\begin{equation}\label{eq:haar-pairing-average}
 B_g(Ph,k)=\frac1{2\pi}\int_0^{2\pi}B_g(R_\theta^*h,k)\,d\theta.
\end{equation}
Isometry invariance of the full pairing gives
$B_g(R_\theta^*h,k)=B_g(h,R_{-\theta}^*k)$.
Inversion of the angle preserves integration over a full period;
symmetry of $B_g$ and \eqref{eq:haar-pairing-average} then give
$B_g(Ph,k)=B_g(h,Pk)$.
Since $P(k-Pk)=0$ by idempotence, self-adjointness implies
$B_g(Ph,k-Pk)=0$. Expand the original quadratic action on
$h=Ph+(h-Ph)$ and use this zero mixed term to obtain
\eqref{eq:haar-action-split}.
\end{proof}

\section{Global classification of invariant one-forms}

The smoothness of the height probes at the poles is essential. The cylinder
coefficients alone do not provide it. We first record the smooth factorization
that removes this difficulty without assuming analyticity.

\begin{lemma}\label{lem:parity-plane-factorization}
Every smooth even function $e:\R\to\R$ has the form $e(t)=E(t^2)$ for some
globally smooth $E$. Every smooth odd function has the form $tF(t^2)$ for
some globally smooth $F$. Consequently, if the smooth pair $(P,Q):\R^2\to\R^2$
is rotation-equivariant, there are globally smooth $A,B:\R\to\R$ such that
\begin{equation}\label{eq:plane-covector-factorization}
 \begin{split}
 P(u,v)&=uA(u^2+v^2)-vB(u^2+v^2),\\
 Q(u,v)&=vA(u^2+v^2)+uB(u^2+v^2).
 \end{split}
\end{equation}
\end{lemma}

\begin{proof}
For a smooth odd function $p$, its Hadamard quotient
$q(t)=\int_0^1p'(st)\,ds$ is smooth and satisfies $p(t)=tq(t)$.
Cancellation for $t\ne0$, and continuity at zero, show that $q$ is even.
For smooth even $e$, apply this observation to the odd function $e'$ and
write $e'(t)=2t\,Te(t)$ with $Te$ smooth and even. On $s>0$,
\[
 \frac{d}{ds}e(\sqrt{s})=Te(\sqrt{s}).
\]
Both sides have continuous limits at zero. The one-sided fundamental theorem
of calculus gives the same derivative identity at zero. Iteration with the
smooth even functions $T^je$ proves that $s\mapsto e(\sqrt{s})$ is smooth
on the closed half-line.

For completeness, this closed-half-line function has a global smooth
extension. If its right-hand jets at zero are $a_j$, choose a smooth compactly
supported cutoff $\eta$ equal to one near zero and take
\[
 G(s)=\sum_{j=0}^\infty \frac{a_j}{j!}s^j\eta(\lambda_j s).
\]
Choose $\lambda_j\geq1$ successively large enough that every derivative
of the $j$th summand of order $k<j$ has supremum norm at most $2^{-j}$.
This is possible since those norms scale as $\lambda_j^{k-j}$.
The series converges in every fixed derivative order and has jets $a_j$.
Joining $G$ on the negative half-line to the given function on the
nonnegative half-line gives a smooth function: at each order the two
one-sided derivatives have the same limit at zero. Applying this construction
to $e(\sqrt{s})$ proves the even factorization, and applying it to the even
quotient $q$ proves the odd factorization.

Restrict the equivariant pair to $(t,0)$. Rotation by $\pi$ makes both
restrictions odd, hence they are $tA(t^2)$ and $tB(t^2)$. For
$\rho=\sqrt{u^2+v^2}>0$, rotate $(\rho,0)$ to $(u,v)$ with cosine
$u/\rho$ and sine $v/\rho$. Equivariance gives
\eqref{eq:plane-covector-factorization}. At the origin, both components
vanish by the same oddness, so the formula holds there as well.
\end{proof}

\begin{lemma}\label{lem:stereographic-invariant-form}
For $\epsilon\in\{1,-1\}$, put $d=4+u^2+v^2$ and define the actual sphere map
\[
 \Psi_\epsilon(u,v)=
 \left(\frac{4u}{d},\frac{4v}{d},
             \frac{\epsilon(4-u^2-v^2)}{d}\right).
\]
It is smooth, intertwines plane rotation with the sphere rotation, and
parametrizes the complement of the pole of height $-\epsilon$.
Its inverse on that complement is
\[
 (x,y,z)\longmapsto
 \left(\frac{2x}{1+\epsilon z},\frac{2y}{1+\epsilon z}\right).
\]
Its actual tangent map is bijective everywhere, and
\begin{equation}\label{eq:stereo-covector-pullbacks}
 \Psi_\epsilon^*dz=-\frac{16\epsilon}{d^2}(u\,du+v\,dv),\qquad
 \Psi_\epsilon^*\kappa=\frac{16}{d^2}(u\,dv-v\,du).
\end{equation}
For every smooth rotation-invariant one-form $h$ on the sphere, there are
globally smooth $A_\epsilon,B_\epsilon:\R\to\R$ for which
\begin{equation}\label{eq:stereo-invariant-form}
 \Psi_\epsilon^*h=
 A_\epsilon(u^2+v^2)(u\,du+v\,dv)
 +B_\epsilon(u^2+v^2)(u\,dv-v\,du).
\end{equation}
\end{lemma}

\begin{proof}
The displayed vector has norm one. Its positive denominator makes it a
smooth sphere-valued map, and direct substitution verifies equivariance
and both inverse laws. Also $1+\epsilon z=8/d$ on its image.
For a plane tangent $(U,V)$, its included derivative is
\[
 \left(\frac{4U}{d}-\frac{8u(uU+vV)}{d^2},
       \frac{4V}{d}-\frac{8v(uU+vV)}{d^2},
       -\frac{16\epsilon(uU+vV)}{d^2}\right).
\]
If this vector is zero, the last component first gives $uU+vV=0$,
and the first two give $U=V=0$. Thus the actual tangent map is injective,
and hence bijective between two-dimensional tangent spaces.
Evaluating $dz$ and $\kappa=-y\,dx+x\,dy$ on it proves
\eqref{eq:stereo-covector-pullbacks}, including at $(0,0)$.

The two coefficients of $\Psi_\epsilon^*h$ are smooth because $h$ and
the actual tangent map are smooth. The chain rule and rotation
invariance of $h$ imply that their coefficient pair is rotation-equivariant:
evaluation on a rotated tangent vector at the rotated point equals
the original evaluation. Lemma~\ref{lem:parity-plane-factorization}
then gives \eqref{eq:stereo-invariant-form} on the entire plane.
\end{proof}

\begin{theorem}\label{thm:invariant-one-form-classification}
For the produced metric $g_D$, every smooth one-form invariant under its
specified circle rotations can be written
\[
 h=M_D(r)+Z_D(s)
\]
with globally smooth real functions $r,s:\R\to\R$. The equality holds
on the entire sphere, including both poles; no equatorial symmetry is assumed.
\end{theorem}

\begin{proof}
First construct global smooth coefficients $P,Q$ with $h=P(z)\,dz+Q(z)\kappa$,
independently of the metric. On the chart in
Lemma~\ref{lem:stereographic-invariant-form}, write $t=u^2+v^2$.
The exact identities
\[
 t=\frac{4(1-\epsilon z)}{1+\epsilon z},\qquad
 \frac{d^2}{16}=\frac{4}{(1+\epsilon z)^2}
\]
convert \eqref{eq:stereo-invariant-form} into local height coefficients
\[
 P_\epsilon(z)=-\frac{4\epsilon}{(1+\epsilon z)^2}
 A_\epsilon\!\left(\frac{4(1-\epsilon z)}{1+\epsilon z}\right),\quad
 Q_\epsilon(z)=\frac{4}{(1+\epsilon z)^2}
 B_\epsilon\!\left(\frac{4(1-\epsilon z)}{1+\epsilon z}\right).
\]
These evaluations determine the original one-form because the actual
tangent map is surjective.

Here is an explicit way to extend and glue these coefficients. Choose a smooth
transition $S$ with $0\leq S\leq1$, $S=0$ on $(-\infty,0]$, $S>0$ on
$(0,\infty)$ and $S=1$ on $[1,\infty)$. The denominator
$\delta(w)=1+wS(4w+3)$ is strictly positive for every real $w$ and equals
$1+w$ for $w\geq-1/2$: for $w\leq-3/4$ it is one, while for
$-3/4<w<0$ it is at least $1+w>1/4$.
Replace $1+\epsilon z$ in the preceding formulas by $\delta(\epsilon z)$.
This produces globally smooth coefficients still representing $h$ on
$\epsilon z\geq-1/2$, including the corresponding pole. Blend the north
and south coefficients with $S(z+1/2)$ and $1-S(z+1/2)$.
On the overlap both pairs represent the same original covector; outside
the overlap the appropriate weight is one. Thus the global smooth $P,Q$
satisfy $h=P(z)\,dz+Q(z)\kappa$ everywhere.

It remains to divide by the metric coefficients without assuming their
positivity outside $[-1,1]$. For any globally smooth $c$ nonzero on that
interval, set
\[
 r_c(v)=\frac{P(v)c(v)}{c(v)^2+S(v^2-1)}.
\]
Its denominator is positive everywhere: on the physical interval $c^2>0$,
and outside it $S(v^2-1)>0$. Hence $r_c$ is globally smooth and $c(v)r_c(v)=P(v)$
on $[-1,1]$. Apply this construction to $(c,P)=(a,P)$ and $(b,Q)$.
The resulting $r,s$ give precisely the already defined sections
$M_D(r)$ and $Z_D(s)$.
\end{proof}

\begin{corollary}\label{cor:haar-zonal-action}
For every smooth one-form $h$ there are globally smooth $r,s:\R\to\R$ with
\[
 Ph=M_D(r)+Z_D(s),\qquad
 Q_{g_D}(h)=2\pi\bigl(J_a[r]+J_a[s]\bigr)+Q_{g_D}(h-Ph).
\]
\end{corollary}

\begin{proof}
The actual smooth average $Ph$ is invariant by
Theorem~\ref{thm:haar-projection}. Apply
Theorem~\ref{thm:invariant-one-form-classification} to that same section,
then use Theorem~\ref{thm:zonal-reduction} and the complete split in
Theorem~\ref{thm:haar-split}. No sign assertion about the remainder is
needed for this identity.
\end{proof}

\section{Scalarization of the complete action}

For a covariant two-tensor $T$, write
\[
 \mathsf A_gT=\frac{T+T^{\mathsf T}}2-\frac{\operatorname{tr}_gT}{2}g.
\]
Thus the already defined one-form operator is $A_gh=\mathsf A_g(\nabla h)$.
On an oriented Riemannian surface let $\Omega$ be its smooth unit area tensor,
normalized by $|\Omega|_g^2=2$. The actual scalar contractions associated with $h$ are
\[
 \tau=\operatorname{tr}_g(\nabla h),\qquad
 \omega=\langle\Omega,\nabla h\rangle_g.
\]
In a positively oriented orthonormal frame, $\omega=\nabla_0h_1-\nabla_1h_0$.
Both functions are smooth, being contractions of the canonical first derivative.

\begin{lemma}\label{lem:surface-tensor-splitting}
For every smooth covariant two-tensor $T$ on an oriented Riemannian surface,
put $t=\operatorname{tr}_gT$ and $w=\langle\Omega,T\rangle_g$. Then
\begin{equation}\label{eq:surface-tensor-split}
 T=\mathsf A_gT+\frac{t}{2}g+\frac{w}{2}\Omega,\qquad
 |T|_g^2=|\mathsf A_gT|_g^2+\frac{t^2+w^2}{2},\qquad
 \langle\mathsf A_gT,T\rangle_g=|\mathsf A_gT|_g^2.
\end{equation}
The genuine covariant derivatives satisfy
\begin{equation}\label{eq:surface-derivative-split}
 |\nabla T|_g^2=|\nabla(\mathsf A_gT)|_g^2+
                 \frac{|dt|_g^2+|dw|_g^2}{2}.
\end{equation}
\end{lemma}

\begin{proof}
At a point choose a positively oriented orthonormal basis and write
$T=\left(\begin{smallmatrix}p&q\\r&s\end{smallmatrix}\right)$.
Then $t=p+s$, $w=q-r$ and
\[
 \mathsf A_gT=\frac12
 \begin{pmatrix}p-s&q+r\\q+r&s-p\end{pmatrix}.
\]
The symmetric trace-free, scalar metric and alternating summands are mutually
orthogonal, with $|g|^2=|\Omega|^2=2$. This proves
\eqref{eq:surface-tensor-split} by direct matrix calculation.

The area tensor is parallel: its derivative remains alternating, hence is
pointwise proportional to $\Omega$; differentiation of its constant squared
norm makes that proportionality coefficient zero, as in the argument of
Lemma~\ref{lem:parallel-area}. Metric compatibility and parallelness therefore give
\[
 dt(X)=\operatorname{tr}_g(\nabla_XT),\qquad
 dw(X)=\langle\Omega,\nabla_XT\rangle_g,
\]
\[
 \nabla_X(\mathsf A_gT)(Y,Z)=
 \frac{(\nabla_XT)(Y,Z)+(\nabla_XT)(Z,Y)}2
       -\frac{dt(X)}2g(Y,Z).
\]
These are the actual tensor derivative rules, including the connection
correction in each covariant slot. Applying the same orthogonal matrix
calculation to $\nabla_XT$ and summing over an orthonormal basis for $X$
proves \eqref{eq:surface-derivative-split}.
\end{proof}

\begin{lemma}\label{lem:canonical-tensor-green}
On a closed Riemannian manifold of positive dimension, let $T$ and $V$ be
smooth covariant tensors of ranks $s$ and $s+1$. Define $\operatorname{div}_gV$
by tracing the first two slots of $\nabla V$. Then
\[
 \int\langle\nabla T,V\rangle_g\,d\mu_g
       =-\int\langle T,\operatorname{div}_gV\rangle_g\,d\mu_g.
\]
Consequently
\[
 \int\langle\nabla T,\nabla U\rangle_g\,d\mu_g
       =-\int\langle T,\Delta^\nabla U\rangle_g\,d\mu_g,
\]
and the actual rough Laplacian is formally self-adjoint.
\end{lemma}

\begin{proof}
Let $X$ be the metric dual of the smooth covector
$Y\mapsto\langle T,V(Y,\cdot)\rangle_g$.
The metric product rule gives
\[
 \operatorname{div}_gX=
       \langle\nabla T,V\rangle_g+
       \langle T,\operatorname{div}_gV\rangle_g.
\]
Every term is smooth and integrable on the compact manifold. Integrate and
use the divergence theorem on the manifold without boundary.
Taking $V=\nabla U$ gives the rough-Laplacian formula. Symmetry of the
metric pairing then gives formal self-adjointness.
\end{proof}

Write $\mathsf S$ for transposition of the first two slots of a covariant
three-tensor: $(\mathsf S B)(X,Y,Z)=B(Y,X,Z)$. This slot permutation is
distinct from the circle-average projector $P$.

\begin{lemma}\label{lem:surface-contracted-commutator}
On a Riemannian surface, every smooth covariant two-tensor $T$ satisfies
\begin{equation}\label{eq:surface-contracted-commutator}
 \operatorname{div}_g(\mathsf S\nabla T)
    =\nabla(\operatorname{div}_gT)+2K_g\mathsf A_gT.
\end{equation}
If the surface is closed, then
\begin{equation}\label{eq:integrated-slot-pairing}
 \int\langle\mathsf S\nabla T,\nabla T\rangle_g\,d\mu_g
   =\int|\operatorname{div}_gT|_g^2\,d\mu_g
       -2\int K_g|\mathsf A_gT|_g^2\,d\mu_g.
\end{equation}
\end{lemma}

\begin{proof}
The curvature convention is
$R(X,Y)Z=K_g(\langle Y,Z\rangle_gX-\langle X,Z\rangle_gY)$.
In an orthonormal frame normal at the point of calculation, the difference
of the two differential operators in \eqref{eq:surface-contracted-commutator} is
\[
 \sum_i([\nabla_i,\nabla_j]T)_{ik}
 =-\sum_i T(R(e_i,e_j)e_i,e_k)-\sum_i T(e_i,R(e_i,e_j)e_k).
\]
The surface curvature formula reduces this to
\[
 K_g\bigl(T_{jk}+T_{kj}-(\operatorname{tr}_gT)g_{jk}\bigr)
       =2K_g(\mathsf A_gT)_{jk}.
\]
Tensoriality makes the result independent of the chosen frame. No symmetry
or trace-free assumption on $T$ was used.

Apply Lemma~\ref{lem:canonical-tensor-green} first with $V=\mathsf S\nabla T$,
then with the lower-rank tensor $\operatorname{div}_gT$ and the tensor $T$.
Substitute \eqref{eq:surface-contracted-commutator}. The first term becomes
$\int|\operatorname{div}_gT|^2$, while
$\langle T,\mathsf A_gT\rangle_g=|\mathsf A_gT|^2$ by
\eqref{eq:surface-tensor-split}. This gives \eqref{eq:integrated-slot-pairing}.
Smoothness and compactness justify splitting the integrals.
\end{proof}

\begin{lemma}\label{lem:one-form-skew-pairing}
For a smooth one-form on a Riemannian surface, its actual second covariant
derivative $B=\nabla^2h$ obeys
\[
 |B|_g^2-\langle\mathsf S B,B\rangle_g=K_g^2|h|_g^2.
\]
\end{lemma}

\begin{proof}
For any three-tensor in an orthonormal two-frame, expansion gives
\[
 \sum_{i,j,k}B_{ijk}^2-\sum_{i,j,k}B_{jik}B_{ijk}
       =(B_{010}-B_{100})^2+(B_{011}-B_{101})^2.
\]
For the genuine second derivative, the covector Ricci identity is
\[
 (\nabla_X\nabla h)(Y,Z)-(\nabla_Y\nabla h)(X,Z)=-h(R(X,Y)Z).
\]
The two differences above are respectively $K_gh(e_1)$ and $-K_gh(e_0)$,
so their squared sum is exactly $K_g^2|h|_g^2$.
\end{proof}

\begin{theorem}\label{thm:complete-scalarization}
On a closed oriented Riemannian surface, every smooth one-form satisfies
the following identity for the original complete action:
\begin{equation}\label{eq:complete-scalarization}
 \begin{split}
 Q_g(h)={}&\int|\nabla(A_gh)|_g^2\,d\mu_g
       +3\int K_g|A_gh|_g^2\,d\mu_g\\
 &+\frac12\left(
       \int\bigl(|d\tau|_g^2-K_g\tau^2\bigr)\,d\mu_g
       +\int\bigl(|d\omega|_g^2-K_g\omega^2\bigr)\,d\mu_g
             \right),
 \end{split}
\end{equation}
where $\tau=\operatorname{tr}_g(\nabla h)$ and
$\omega=\langle\Omega,\nabla h\rangle_g$ are the actual contractions.
All terms are integrable. No curvature-sign or pinching hypothesis is required.
\end{theorem}

\begin{proof}
Set $N=\nabla h$ and $B=\nabla N$. The divergence of $N$ is the actual
rough Laplacian $\Delta^\nabla h$. Combining
Lemmas~\ref{lem:surface-contracted-commutator} and
\ref{lem:one-form-skew-pairing} gives
\begin{equation}\label{eq:second-derivative-energy}
 \int\bigl(|\Delta^\nabla h|_g^2+K_g^2|h|_g^2\bigr)\,d\mu_g
     =\int|B|_g^2\,d\mu_g+2\int K_g|A_gh|_g^2\,d\mu_g.
\end{equation}
Substitution into \eqref{eq:complete-dissipation} yields
\[
 Q_g(h)=\int|B|_g^2\,d\mu_g
           -\int K_g|N|_g^2\,d\mu_g
           +4\int K_g|A_gh|_g^2\,d\mu_g.
\]
The two pointwise identities from Lemma~\ref{lem:surface-tensor-splitting} are
\[
 |N|_g^2=|A_gh|_g^2+\frac{\tau^2+\omega^2}{2},\qquad
 |B|_g^2=|\nabla(A_gh)|_g^2+\frac{|d\tau|_g^2+|d\omega|_g^2}{2}.
\]
Inserting them proves \eqref{eq:complete-scalarization}.
The original fields, their canonical derivatives, the curvature and the
scalar contractions are smooth. Thus every displayed integrand is continuous
and integrable on the compact surface, justifying all additions and subtractions.
\end{proof}

\begin{corollary}\label{cor:rotational-scalarization}
For the produced metric $g_D$, use the actual area tensor $\Omega_D$ from
Lemma~\ref{lem:parallel-area}, and define
$\tau_D=\operatorname{tr}_{g_D}(\nabla h)$ and
$\omega_D=\langle\Omega_D,\nabla h\rangle_{g_D}$. Every smooth one-form obeys
\[
 \begin{split}
 Q_{g_D}(h)={}&\int|\nabla(A_{g_D}h)|_{g_D}^2\,d\mu_{g_D}
        +3\int\frac{|A_{g_D}h|_{g_D}^2}{a(z)}\,d\mu_{g_D}\\
 &+\frac12\left(
    \int\left(|d\tau_D|_{g_D}^2-\frac{\tau_D^2}{a(z)}\right)\,d\mu_{g_D}
   +\int\left(|d\omega_D|_{g_D}^2-\frac{\omega_D^2}{a(z)}\right)\,d\mu_{g_D}
              \right).
 \end{split}
\]
\end{corollary}

\begin{proof}
The constructed $\Omega_D$ is smooth, alternating and of squared norm two.
Theorem~\ref{thm:curvature-profile} gives the same metric's curvature
$K_{g_D}=1/a(z)$ everywhere, including both poles. Apply
Theorem~\ref{thm:complete-scalarization} to that metric, that area tensor,
and the original form $h$.
\end{proof}

\section{Scalar Haar compatibility}

We use the same rotations $R_\theta$, metric $g_D$ and area tensor $\Omega_D$
as above. For a scalar function $u$ on the sphere, put
\begin{equation}\label{eq:scalar-haar-average}
 P_0u(x)=\frac{1}{2\pi}\int_0^{2\pi}u(R_\theta x)\,d\theta.
\end{equation}
This scalar operator is distinguished from the covector pullback average $P$.

\begin{lemma}\label{lem:rotation-area-invariance}
Every rotation of the actual circle action preserves the constructed area tensor:
$R_\theta^*\Omega_D=\Omega_D$ on the whole sphere.
\end{lemma}

\begin{proof}
Write $L_\theta$ for the ambient orthogonal rotation fixing the third coordinate.
Its matrix has determinant one: its horizontal block has determinant
$\cos^2\theta+\sin^2\theta=1$ and its vertical entry is one.
The derivative of $R_\theta$ is the restriction of $L_\theta$ to tangent vectors.
The defining ambient formula therefore gives
\[
 \begin{split}
 (R_\theta^*\Omega_D)_x(V,W)
 &=a(z(R_\theta x))
       \det(L_\theta x,L_\theta V,L_\theta W)\\
 &=a(z(x))\det(x,V,W)=(\Omega_D)_x(V,W).
 \end{split}
\]
Here tangent vectors are identified with their actual ambient inclusion.
The computation also holds at either pole, and equally for every unit complex
number representing a circle rotation.
\end{proof}

\begin{lemma}\label{lem:scalar-haar-projector}
The operator $P_0$ maps smooth scalars to smooth scalars, is real linear on
continuous scalars, fixes constants, and satisfies
\[
 P_0u(R_\phi x)=P_0u(x),\qquad
 P_0(u\circ R_\phi)=P_0u,\qquad P_0(P_0u)=P_0u.
\]
At either pole $p$, $P_0u(p)=u(p)$. Consequently a scalar with zero angular
average vanishes at both poles.
\end{lemma}

\begin{proof}
Joint smoothness of $(\theta,x)\mapsto R_\theta x$ makes
$(\theta,x)\mapsto u(R_\theta x)$ smooth when $u$ is smooth. In any relatively
compact coordinate neighborhood, every derivative with respect to $x$ is
bounded on the product with the compact angle interval. Differentiation under
the integral thus proves smoothness of every order. For continuous inputs the
angle integrands are genuinely integrable, giving linearity. The normalization
makes the average of a constant equal to that constant.

The group law changes either rotation of the input or rotation of the base
point into a shift of the angle variable. Periodicity makes the integral over
one full period unchanged. The resulting averaged function is invariant,
so averaging it again integrates a constant angle orbit and proves idempotence.
Every rotation fixes either pole, whence its scalar orbit there is the constant
$u(p)$. This proves the last assertion without division by a polar weight.
\end{proof}

\begin{theorem}\label{thm:scalar-haar-compatibility}
For the original smooth one-form $h$, let
$\tau(h)=\operatorname{tr}_{g_D}(\nabla h)$ and
$\omega(h)=\langle\Omega_D,\nabla h\rangle_{g_D}$. Then
\[
 \tau(Ph)=P_0\tau(h),\qquad \omega(Ph)=P_0\omega(h).
\]
In particular, for $h_\perp=h-Ph$,
\begin{equation}\label{eq:remainder-scalar-means}
 P_0\tau(h_\perp)=0,\qquad P_0\omega(h_\perp)=0.
\end{equation}
Both actual scalars $\tau(h_\perp)$ and $\omega(h_\perp)$ vanish at both poles.
No curvature pinching or invariance hypothesis on $h$ is imposed.
\end{theorem}

\begin{proof}
First recall why the scalar contractions have the required naturality. If $F$
is a diffeomorphism, an orthonormal frame for $F^*g$ is taken by $dF$ to an
orthonormal frame for $g$. Computing contractions in these two frames proves
\[
 \operatorname{tr}_{F^*g}(F^*T)=(\operatorname{tr}_gT)\circ F,
 \qquad
 \langle F^*S,F^*T\rangle_{F^*g}=\langle S,T\rangle_g\circ F.
\]
The already established canonical derivative naturality identifies
$\nabla^{F^*g}(F^*h)$ with $F^*(\nabla^g h)$, including the derivative of $F$
in both slots. These statements hold with no orientation restriction when the
area tensor is also pulled back. For the present rotations, both $g_D$ and
$\Omega_D$ are preserved, the latter by Lemma~\ref{lem:rotation-area-invariance}.
It follows that
\[
 \tau(R_\theta^*h)=\tau(h)\circ R_\theta,
 \qquad \omega(R_\theta^*h)=\omega(h)\circ R_\theta.
\]

The tensor average commutes with the actual covariant derivative. At each
fixed base point, trace and contraction against the fixed area tensor are
continuous linear maps on a finite-dimensional tensor fiber. They commute
with its integral. All angle integrands are continuous on a compact interval,
so this interchange is justified by genuine integrability. Taking those two
contractions of $\nabla(Ph)=P(\nabla h)$ proves the displayed compatibility
identities with exactly the normalization in \eqref{eq:scalar-haar-average}.

The actual complementary form satisfies $P(h-Ph)=0$. Apply compatibility to
that form; the covariant derivative of the zero section is zero, so both scalar
averages vanish. Finally apply Lemma~\ref{lem:scalar-haar-projector} at the poles.
\end{proof}

The pole values in this section are the first endpoint facts needed for
the scalar energy estimate. We now establish the additional bounds that justify
division by the vanishing polar weight.

\section{The positive scalar Haar estimate}

\begin{lemma}\label{lem:polar-scalar-geometry}
The actual map
\[
 \Phi(s,\theta)=(\sin s\cos\theta,\sin s\sin\theta,\cos s)
\]
is smooth on $\R^2$ with values in the unit sphere, including at $s=0,\pi$.
Its derivatives $V_s=\partial_s\Phi$ and $V_\theta=\partial_\theta\Phi$ satisfy
\[
 g_D(V_s,V_s)=\frac{a(\cos s)^2}{b(\cos s)},\qquad
 g_D(V_\theta,V_\theta)=b(\cos s)\sin^2s,\qquad
 g_D(V_s,V_\theta)=0.
\]
They form a tangent basis when $\sin s\ne0$. For every smooth scalar $u$, writing
$y(s,\theta)=u(\Phi(s,\theta))$, its actual differential therefore obeys
\begin{equation}\label{eq:polar-scalar-norm}
 |du|_{g_D}^2(\Phi(s,\theta))
 =\frac{b(\cos s)}{a(\cos s)^2}y_s^2
       +\frac{y_\theta^2}{b(\cos s)\sin^2s}
 \qquad(\sin s\ne0).
\end{equation}
For every continuous scalar $F$ on the sphere,
\begin{equation}\label{eq:polar-metric-integration}
 \int F\,d\mu_{g_D}
   =\int_0^\pi\int_0^{2\pi}
       a(\cos s)\sin s\,F(\Phi(s,\theta))\,d\theta\,ds.
\end{equation}
\end{lemma}

\begin{proof}
The displayed ambient map has norm one and is smooth, so its restriction to
the sphere is a genuine smooth map. Direct differentiation gives
\[
 V_s=(\cos s\cos\theta,\cos s\sin\theta,-\sin s),\quad
 V_\theta=(-\sin s\sin\theta,\sin s\cos\theta,0).
\]
Their round Gram matrix is $\operatorname{diag}(1,\sin^2s)$, while
$dz(V_s)=-\sin s$ and $dz(V_\theta)=0$. Substitute these into the original
tensor $b(z)g_{\rm round}+(c(z)/b(z))dz\otimes dz$ and use
$a^2-b^2=(1-z^2)c$. This proves the three metric evaluations even at the poles.
Away from the poles, the round Gram matrix proves independence, and dimension
two gives spanning. Inverting the resulting metric Gram matrix and applying
the chain rule to $du(V_s)$ and $du(V_\theta)$ proves
\eqref{eq:polar-scalar-norm}.

For completeness, the volume formula follows without treating the polar map
as a global one-to-one chart. Rotations preserve $g_D$ and its volume. Fubini,
justified by continuity on the compact angle interval times the compact sphere,
therefore gives $\int P_0F\,d\mu_{g_D}=\int F\,d\mu_{g_D}$.
Every point $x$ lies on the orbit of $\Phi(\arccos z(x),0)$: choose the argument
of its horizontal complex coordinate. The same assertion holds at the poles
because their horizontal coordinate is zero. Thus the invariant average is
the continuous height function
$H(v)=P_0F(\Phi(\arccos v,0))$ evaluated at $v=z(x)$.
The established height disintegration gives $2\pi\int_{-1}^1 a(v)H(v)\,dv$.
Substituting $v=\cos s$ and using the literal normalization of $P_0$ proves
\eqref{eq:polar-metric-integration}. All integrands used in this argument are
continuous on their compact domains.
\end{proof}

\begin{lemma}\label{lem:angular-wirtinger}
If $f\in C^1(\R)$, $f(2\pi)=f(0)$ and $\int_0^{2\pi}f=0$, then
\[
 \int_0^{2\pi}f^2\,d\theta\leq\int_0^{2\pi}(f')^2\,d\theta.
\]
\end{lemma}

\begin{proof}
With normalized Fourier coefficients
$\widehat f(n)=(2\pi)^{-1}\int_0^{2\pi}f(\theta)e^{-in\theta}\,d\theta$,
integration by parts and the matching endpoint values give
$\widehat{f'}(n)=in\widehat f(n)$. The zero mean removes $n=0$.
Parseval's identity for $L^2$ functions on the circle
\cite[Corollary 2, p.~5]{jerison-fourier} gives
\[
 \frac1{2\pi}\int f^2
   =\sum_{n\ne0}|\widehat f(n)|^2
   \leq\sum_{n\in\mathbb Z}|\widehat{f'}(n)|^2
   =\frac1{2\pi}\int(f')^2.
\]
Both functions are in $L^2$ because they are continuous on the compact interval.
Only the endpoint values of $f$ need agree; no endpoint condition on $f'$ is
required. The Fourier identities concern the interval representatives of the
corresponding $L^2$ functions on the circle.
\end{proof}

\begin{lemma}\label{lem:polar-endpoint-energies}
Let $y\in C^2(\R^2)$ satisfy $y(0,\theta)=y(\pi,\theta)=0$ for every $\theta$.
On each compact angular interval there is $C>0$ such that, for $0\leq s\leq\pi$,
\[
 |y|\leq C\sin s,\qquad |y_s|\leq C,\qquad |y_\theta|\leq C\sin s.
\]
If $w:\R\to\R$ is continuous and positive on $[0,\pi]$ and $W(s)=\sin s\,w(s)$, the
following functions are integrable on every such polar rectangle:
\[
 Wy_s^2,\quad \frac{y^2}{W},\quad \frac{y_\theta^2}{W},\quad
 \frac{(Wy_s-\cos s\,y)^2}{W},\quad
 -\sin s\,y^2+2\cos s\,yy_s,\quad
 \frac{\sin^2s\,y^2}{W},\quad \sin s\,y^2,\quad \cos s\,y^2.
\]
The quotients are considered on the open radial interval; their assigned
endpoint values do not affect the integrals. The boundary primitive
$\cos s\,y(s,\theta)^2$ tends to zero at both ends.
\end{lemma}

\begin{proof}
The first radial derivative and the mixed radial-angular derivative are
bounded on the compact rectangle. The angular derivative vanishes at both
radial endpoints because the two boundary slices are identically zero.
Apply the mean value estimate from the nearer endpoint to $y$ and to
$y_\theta$. On the northern half, $s\leq(\pi/2)\sin s$; on the southern half,
$\pi-s\leq(\pi/2)\sin s$. Enlarging one positive constant proves all three
uniform bounds.

Let $m=\min_{[0,\pi]}w>0$. For either $v=y$ or $v=y_\theta$, the bound
$|v|\leq C\sin s$ gives
\[
 0\leq\frac{v^2}{\sin s\,w(s)}\leq\frac{C^2}{m}
 \qquad(0<s<\pi).
\]
These measurable quotients are therefore integrable on the finite rectangle.
The radial, potential and boundary terms are continuous and integrable.
Multiplication of $y^2/W$ by the bounded factors $\sin^2s$ and $\cos^2s$
preserves integrability. Finally,
\[
 \frac{(Wy_s-\cos s\,y)^2}{W}
     =Wy_s^2+\cos^2s\,\frac{y^2}{W}-2\cos s\,yy_s
\]
proves integrability of the completed square. Continuity and the zero boundary
values give the stated limits; the uniform bound also controls the boundary
primitive by $C^2\sin^2s$.
\end{proof}

\begin{lemma}\label{lem:polar-square-completion}
Let $y\in C^1(\R)$ vanish at $0$ and $\pi$. Suppose $W>0$ on $(0,\pi)$ and
both $Wy'^2$ and $y^2/W$ are integrable there. Then the square term below is
integrable and
\begin{equation}\label{eq:polar-square-completion}
 \int_0^\pi\left(Wy'^2+\frac{y^2}{W}-\sin s\,y^2\right)ds
 =\int_0^\pi\frac{(Wy'-\cos s\,y)^2}{W}\,ds
   +\int_0^\pi\frac{\sin^2s\,y^2}{W}\,ds.
\end{equation}
No derivative or endpoint positivity of $W$ is required.
\end{lemma}

\begin{proof}
Expansion of the square and $\sin^2s+\cos^2s=1$ give the pointwise identity
\[
 Wy'^2+\frac{y^2}{W}-\sin s\,y^2
 =\frac{(Wy'-\cos s\,y)^2}{W}
   +\frac{\sin^2s\,y^2}{W}+(\cos s\,y^2)'.
\]
The expanded square is integrable by the two finite-energy hypotheses and
continuity of $y,y'$. The last derivative is
$-\sin s\,y^2+2\cos s\,yy'$, a continuous integrable function.
Its integral is zero by the fundamental theorem of calculus and the two zero
boundary values. This proves the identity and its positive lower bound after
dropping the square.
\end{proof}

\begin{theorem}\label{thm:positive-polar-gap}
Under the hypotheses of Lemma~\ref{lem:polar-endpoint-energies}, suppose also
that $y(s,2\pi)=y(s,0)$ and $\int_0^{2\pi}y(s,\theta)\,d\theta=0$ for every $s$.
Then
\[
 \int_0^\pi\int_0^{2\pi}\frac{\sin^2s\,y^2}{W}\,d\theta\,ds
 \leq\int_0^\pi\int_0^{2\pi}
    \left(Wy_s^2+\frac{y_\theta^2}{W}-\sin s\,y^2\right)d\theta\,ds.
\]
\end{theorem}

\begin{proof}
All terms are product integrable by Lemma~\ref{lem:polar-endpoint-energies}.
Consequently almost every radial slice has the finite energies needed for
Lemma~\ref{lem:polar-square-completion}. Integrate its lower bound in $\theta$
and apply Fubini. For every interior radial point, Lemma~\ref{lem:angular-wirtinger}
and $W(s)>0$ give $\int y^2/W\leq\int y_\theta^2/W$ in the angular variable.
Integrating this comparison in $s$ and adding the common radial and potential
terms proves the claim. All slice and integral exchanges use the previously
proved product integrability.
\end{proof}

\begin{theorem}\label{thm:positive-scalar-haar-gap}
For the original produced metric $g_D$, every smooth scalar $u$ with $P_0u=0$
satisfies the positive estimate
\begin{equation}\label{eq:positive-scalar-haar-gap}
 \int\left(|du|_{g_D}^2-K_{g_D}u^2\right)d\mu_{g_D}
       \geq\int\frac{u^2}{b(z)}\,d\mu_{g_D}.
\end{equation}
All polar endpoint and integrability conditions follow from these hypotheses.
No curvature pinching is assumed.
\end{theorem}

\begin{proof}
Set $y=u\circ\Phi$ and $w(s)=b(\cos s)/a(\cos s)$.
The actual global polar map makes $y$ smooth on $\R^2$.
The condition $P_0u=0$ gives both zero pole values by
Lemma~\ref{lem:scalar-haar-projector}; hence $y$ has zero radial boundary slices.
Rotation compatibility of $\Phi$ gives the angular period and zero angular
integrals directly from the literal definition of $P_0$.
Positivity and continuity of $a,b$ on the physical height interval make $w$
continuous and positive, with a positive minimum on $[0,\pi]$.
Thus every hypothesis, bound and integrability conclusion of
Theorem~\ref{thm:positive-polar-gap} applies.

By Lemma~\ref{lem:polar-scalar-geometry}, the Dirichlet density times the volume
Jacobian is $Wy_s^2+y_\theta^2/W$, where $W=\sin s\,b(\cos s)/a(\cos s)$.
The actual curvature identity $K_{g_D}=1/a(z)$ changes its potential density
into $\sin s\,y^2$. The right side of \eqref{eq:positive-scalar-haar-gap} has
polar density $a(\cos s)\sin s\,y^2/b(\cos s)=\sin^2s\,y^2/W$.
The exact volume formula and Theorem~\ref{thm:positive-polar-gap} therefore give
\eqref{eq:positive-scalar-haar-gap}. The intrinsic densities are continuous on
the compact sphere, and the polar densities are integrable by the established
endpoint estimates. No expression is evaluated by division at a geometric pole.
\end{proof}

\section{Strict positivity of the nonzonal remainder}

\begin{lemma}\label{lem:weighted-energy-rigidity}
Let $(M,g)$ be a compact smooth Riemannian manifold, let $w:M\to\R$ be
continuous and strictly positive, and let $T$ be a smooth covariant tensor.
Then
\[
 \int_M w|T|_g^2\,d\mu_g=0\quad\Longleftrightarrow\quad T=0.
\]
Likewise, for every continuous real scalar $v$,
$\int_M wv^2\,d\mu_g=0$ if and only if $v=0$.
\end{lemma}

\begin{proof}
The Riemannian density is positive in every coordinate neighborhood, so every
nonempty open set has positive volume. The displayed integrands are continuous,
nonnegative and integrable by compactness. A zero integral makes the integrand
zero almost everywhere. Continuity and positivity of the measure on open sets
then make it zero everywhere: a positive value would persist on a nonempty
open neighborhood and contradict almost-everywhere vanishing. Strict positivity
of $w$ and positive definiteness of the tensor norm give $T=0$. The scalar case
uses the same argument with $v^2$. Both converse implications are immediate.
\end{proof}

\begin{theorem}\label{thm:nonzonal-positivity-rigidity}
For every produced rotational metric $g_D$ and smooth one-form $h$, put
$k=h-Ph$, $A=A_{g_D}k$, $\tau=\operatorname{tr}_{g_D}(\nabla k)$ and
$\omega=\langle\Omega_D,\nabla k\rangle_{g_D}$. The original complete action satisfies
\begin{equation}\label{eq:nonzonal-positive-budget}
 Q_{g_D}(k)\geq
 3\int K_{g_D}|A|_{g_D}^2\,d\mu_{g_D}
 +\frac12\left(\int\frac{\tau^2}{b(z)}\,d\mu_{g_D}
                 +\int\frac{\omega^2}{b(z)}\,d\mu_{g_D}\right)\geq0.
\end{equation}
Moreover,
\begin{equation}\label{eq:nonzonal-zero-rigidity}
 Q_{g_D}(h-Ph)=0\quad\Longleftrightarrow\quad h-Ph=0.
\end{equation}
Thus every nonzero Haar remainder has strictly positive action, with no
curvature pinching assumption.
\end{theorem}

\begin{proof}
The actual scalars $\tau,\omega$ are smooth and have zero angular mean by
Theorem~\ref{thm:scalar-haar-compatibility}. Apply
Theorem~\ref{thm:positive-scalar-haar-gap} to each. The literal scalarization
in Corollary~\ref{cor:rotational-scalarization} gives
\[
 Q_{g_D}(k)=\int|\nabla A|_{g_D}^2\,d\mu_{g_D}
        +3\int K_{g_D}|A|_{g_D}^2\,d\mu_{g_D}
        +\frac12\left(E(\tau)+E(\omega)\right),
\]
where $E(v)=\int(|dv|_{g_D}^2-K_{g_D}v^2)\,d\mu_{g_D}$ denotes the scalar
energy appearing in that identity. The two proved scalar bounds and the
nonnegative first integral prove \eqref{eq:nonzonal-positive-budget}.
Here $K_{g_D}=1/a(z)>0$ and $b(z)>0$ on the entire sphere.

Suppose $Q_{g_D}(k)=0$. All three weighted integrals in the lower bound must
vanish. Apply Lemma~\ref{lem:weighted-energy-rigidity} with weights $K_{g_D}$
and $1/b(z)$ to obtain $A=0$, $\tau=0$ and $\omega=0$ everywhere.
The actual tensor decomposition from Lemma~\ref{lem:surface-tensor-splitting}
then gives
\[
 \nabla k=A+\frac{\tau}{2}g_D+\frac{\omega}{2}\Omega_D=0.
\]
This is an equality of the original smooth tensor sections. Differentiating it
gives $\nabla^2k=0$, and tracing gives $\Delta^\nabla k=0$.
Substitution in the original four-term action \eqref{eq:complete-dissipation},
not merely its scalar lower bound, leaves
\[
 0=Q_{g_D}(k)=\int K_{g_D}^2|k|_{g_D}^2\,d\mu_{g_D}.
\]
The weight $K_{g_D}^2$ is continuous and strictly positive. A final application
of Lemma~\ref{lem:weighted-energy-rigidity} gives $k=0$.
Conversely the canonical derivatives of the zero section and all four original
density terms vanish, so its action is zero. Nonnegativity and the zero
classification give strict positivity when $k\ne0$.
\end{proof}

\begin{corollary}\label{cor:haar-action-comparison}
For every smooth one-form on the same produced metric,
\[
 Q_{g_D}(Ph)\leq Q_{g_D}(h).
\]
Equality holds exactly when $h=Ph$, equivalently when $h$ is invariant under
every rotation of the actual circle action.
\end{corollary}

\begin{proof}
The complete Haar split of Theorem~\ref{thm:haar-split} gives
$Q_{g_D}(h)-Q_{g_D}(Ph)=Q_{g_D}(h-Ph)$.
Apply Theorem~\ref{thm:nonzonal-positivity-rigidity}. The fixed points of the
actual projector are precisely the invariant forms, by
Theorem~\ref{thm:haar-projection}.
\end{proof}

\begin{corollary}\label{cor:strict-zonal-bound}
For every smooth $h$ there are globally smooth real probes $r,s$ such that
\[
 Ph=M_D(r)+Z_D(s),\qquad
 Q_{g_D}(h)\geq2\pi\bigl(J_a[r]+J_a[s]\bigr).
\]
For these same probes, equality holds if and only if
$h=M_D(r)+Z_D(s)$.
\end{corollary}

\begin{proof}
Use the original invariant-form classification to represent the actual $Ph$.
The exact zonal identity gives
$Q_{g_D}(Ph)=2\pi(J_a[r]+J_a[s])$ for those probes. Both the bound and its
equality case follow from Corollary~\ref{cor:haar-action-comparison}.
The global smoothness and the section equality retain their validity at both
poles; no equatorial symmetry is assumed.
\end{proof}

\section{Hemisphere substitution and conformal-Killing safety}

Write $K_{g_D}=\operatorname{Scal}_{g_D}/2$ and retain the original
$H=f/a$, sections $M_D,Z_D$, projector $P$ and complete action $Q_{g_D}$.
A smooth one-form is conformal Killing, abbreviated CK, when its actual
Ahlfors derivative vanishes:
$A_{g_D}h=\operatorname{sym}(\nabla h)
-\tfrac12\operatorname{tr}_{g_D}(\nabla h)g_D=0$.

\begin{lemma}\label{lem:hemisphere-pair-substitution}
Suppose $0<m\leq a(v)\leq M$ on $[-1,1]$ and put $C=M/m$. The functions
\[
 k_+(v)=\log\frac{M}{a(v)},\qquad
 k_-(v)=\log\frac{M}{a(-v)}
\]
are continuous on $[0,1]$, differentiable on $(0,1)$ and admissible at cap $C$.
Their derivatives are $-a'(v)/a(v)$ and $a'(-v)/a(-v)$, respectively.
Moreover,
\begin{equation}\label{eq:hemisphere-pair-substitution}
 \I(k_+)=\int_0^1 H(v)\,dv,\qquad
 \I(k_-)=\int_0^1 H(-v)\,dv,\qquad
 \I(k_+)+\I(k_-)=\int_{-1}^1H(v)\,dv.
\end{equation}
No equatorial symmetry is assumed.
\end{lemma}

\begin{proof}
The bounds imply $M>0$, $C\geq1$ and $1\leq M/a(\pm v)\leq C$.
Monotonicity of the logarithm gives both obstacle bounds. Positivity on the
physical interval, smoothness of $a$ and the chain rule give the stated
regularity and derivatives; global positivity outside that interval is unnecessary.
Fubini on the triangular domain in the definition of $\I$ gives, for every
continuous $k$ on $[0,1]$,
\[
 \I(k)=\int_0^1 e^{k(v)}
                  \left(\int_v^1 2s e^{-k(s)}\,ds\right)dv .
\]
The integrand is bounded on the compact square, which supplies genuine
integrability before interchanging integrals. For $k_+$ its inner expression
becomes $2\int_v^1s a(s)\,ds/a(v)=H(v)$.

For the south, let $\widetilde a(s)=a(-s)$. Changing variables and using
the original balance condition gives
\[
 2\int_v^1s\widetilde a(s)\,ds
 =-2\int_{-1}^{-v}t a(t)\,dt
 =2\int_{-v}^1t a(t)\,dt=f(-v).
\]
Thus its quotient is $H(-v)$. Splitting the full height integral at zero
and reflecting the negative interval proves the last identity.
All quotients are continuous on their compact integration intervals.
\end{proof}

\begin{lemma}\label{lem:constant-zonal-coefficient}
For every $M>0$ and real constants $c,d$,
\begin{equation}\label{eq:constant-zonal-pair-coefficient}
 Q_{g_D}\bigl(M_D(c)+Z_D(d)\bigr)
 =2\pi\left(2\bigl(\I(k_+)+\I(k_-)\bigr)-\frac43\right)(c^2+d^2),
\end{equation}
where $k_\pm(v)=\log(M/a(\pm v))$. Here constant real numbers denote
the corresponding globally constant smooth probes.
\end{lemma}

\begin{proof}
For a constant probe, the actual operator in
\eqref{eq:meridional-operator} is $L_a c=-c$. Its action density in
\eqref{eq:meridional-action} is therefore $(2H(v)-2v^2)c^2$.
The compact-interval continuity of $H$ supplies integrability, and
$\int_{-1}^1v^2\,dv=2/3$ yields
\[
 J_a[c]=\left(2\int_{-1}^1H(v)\,dv-\frac43\right)c^2.
\]
Apply the exact zonal formula of Theorem~\ref{thm:zonal-reduction} and the
hemisphere identities proved above. Those identities only need $M>0$;
the additional $m,M$ bounds in Lemma~\ref{lem:hemisphere-pair-substitution}
are used for admissibility, not this equality.
\end{proof}

\begin{theorem}\label{thm:critical-pair-cap}
Define
\begin{equation}\label{eq:critical-pair-cap}
 C_{\mathrm{CK}}
 =\left(\frac{5+\sqrt{13}}3\right)^{1/3}
       \exp\left(\frac{4+2\sqrt{13}}9\right)>1.
\end{equation}
For the actual optimal value
$m(C)=\I(k_{*,C})=2/3-\beta(C)+1/(3\beta(C))$ from
Theorem~\ref{thm:minimum}, one has
\[
 m(C_{\mathrm{CK}})=\frac13,\qquad
 m(C)\geq\frac13\quad\Longleftrightarrow\quad C\leq C_{\mathrm{CK}}
 \quad(C\geq1).
\]
Every admissible profile continuous on $[0,1]$ and differentiable on $(0,1)$
at a cap $1\leq C\leq C_{\mathrm{CK}}$ satisfies $\I(k)>1/3$.
\end{theorem}

\begin{proof}
Since $\beta>0$, the equation $m(C)=1/3$ is equivalent to
$3\beta^2-\beta-1=0$. Its positive root is
$\beta_*=(1+\sqrt{13})/6$. The identity
$\beta^2=q^3/(q^3+2)$ from \eqref{eq:matching} gives
$q_*^3=(5+\sqrt{13})/3$. Substitution in the actual cap equation
\eqref{eq:cap} yields \eqref{eq:critical-pair-cap}.
Conversely this positive $q_*>1$ satisfies the cap equation for that
displayed number. Uniqueness of the canonical cap parameter therefore
identifies it with $q_*$ and proves the claimed critical value.

The canonical parameter increases strictly with $C$.
The function $q\mapsto q^3/(q^3+2)$ increases strictly for $q>0$, so the
positive upper contact also increases strictly. Both terms
$-\beta$ and $1/(3\beta)$ decrease strictly. Thus the actual $m(C)$ is
strictly decreasing, proving the equivalence around its critical value.
For $C>1$, the contact derivative jumps in Theorem~\ref{thm:contacts}
and optimizer uniqueness exclude equality for a differentiable competitor;
hence $\I(k)>m(C)\geq1/3$.
For $C=1$, the obstacle box forces $k=0$ on $[0,1]$ and the directly
computed value is $2/3>1/3$. This branch does not use a contact jump.
\end{proof}

\begin{theorem}\label{thm:invariant-ck-classification}
For $g_D$, every constant zonal form $M_D(c)+Z_D(d)$ is CK.
A smooth invariant one-form is CK if and only if it has this representation
for real constants $c,d$. The actual projector preserves CK, and therefore
every smooth CK form $h$ satisfies
\[
 Ph=M_D(c)+Z_D(d)
\]
for some real constants. The original unit meridional form $M_D(1)$ is
nonzero and CK.
\end{theorem}

\begin{proof}
The CK equation is linear. The actual first derivative, metric trace and
Ahlfors projection commute with derivative pullback through an isometry,
by Lemma~\ref{lem:pullback-dissipation}.
The meridional derivative formula in Lemma~\ref{lem:meridional-derivatives}
gives
\[
 |A_{g_D}M_D(r)|^2=\frac12(Hr')^2.
\]
Parallel Hodge rotation preserves the actual Ahlfors norm, so the same
formula holds for $Z_D(r)$. Constant probes thus give CK forms.

Apply Theorem~\ref{thm:invariant-one-form-classification} to an invariant
smooth $h$, obtaining $h=M_D(r)+Z_D(s)$ with globally smooth probes.
Meridian reflection is an actual isometry fixing $M_D(r)$ and negating
$Z_D(s)$. Half the sum and half the difference of $h$ and its reflected
pullback are therefore CK, so each summand is CK.
At every height $v\in(-1,1)$, use a genuine cylinder point in the norm
formula above. Since $H(v)>0$, it forces $r'(v)=s'(v)=0$.
The mean-value theorem makes both probes constant on the open interval,
and continuity includes its endpoints. Their values outside $[-1,1]$
are irrelevant to the section, so this gives exactly
$h=M_D(r(0))+Z_D(s(0))$, without claiming global constancy of the extensions.
Conversely these constant sums are invariant by the actual rotation laws
and CK by the first part.

The actual derivative and Ahlfors intertwining in
Theorem~\ref{thm:haar-derivatives} sends $A_{g_D}h=0$ to
$A_{g_D}(Ph)=0$. The average is invariant, so its classification applies.
The previously proved equatorial evaluation of $M_D(1)$ is nonzero;
the constant-probe calculation proves its CK equation.
\end{proof}

\begin{lemma}\label{lem:curvature-profile-box}
For positive real numbers $\kappa,\Lambda$, the actual curvature bounds
$\kappa\leq K_{g_D}(x)\leq\Lambda$ for every sphere point $x$ are equivalent to
\[
 \frac1\Lambda\leq a(v)\leq\frac1\kappa
 \qquad(-1\leq v\leq1).
\]
\end{lemma}

\begin{proof}
Theorem~\ref{thm:curvature-profile} gives $K_{g_D}=1/a(z)$ at every point,
including both poles. Each height $v\in[-1,1]$ is realized by the actual
polar point $\Phi(\arccos v,0)$. The result follows in both directions
by taking reciprocals of positive quantities. No coordinate restriction
removes a pole.
\end{proof}

\begin{theorem}\label{thm:produced-ck-safety}
Let $g_D$ be an original produced metric and $h$ any smooth CK one-form.
If $0<m\leq a\leq M$ on $[-1,1]$ and $M/m\leq C_{\mathrm{CK}}$, then
\begin{equation}\label{eq:produced-ck-safety}
 Q_{g_D}(h)\geq0,\qquad Q_{g_D}(h)=0\quad\Longleftrightarrow\quad h=0.
\end{equation}
The same conclusions hold if $\kappa>0$,
$\kappa\leq K_{g_D}\leq\Lambda$ everywhere and
$\Lambda/\kappa\leq C_{\mathrm{CK}}$.
\end{theorem}

\begin{proof}
Use Lemma~\ref{lem:hemisphere-pair-substitution} at $C=M/m$.
Both actual hemisphere logarithms are differentiable admissible competitors.
Theorem~\ref{thm:critical-pair-cap} gives
$\I(k_+)>1/3$ and $\I(k_-)>1/3$, including when $C=1$.
Consequently the exact coefficient
$2(\I(k_+)+\I(k_-))-4/3$ in
\eqref{eq:constant-zonal-pair-coefficient} is strictly positive.
Every constant zonal sum therefore has nonnegative action; if its action
is zero, then $c^2+d^2=0$, hence $c=d=0$ and the section is zero.

By Theorem~\ref{thm:invariant-ck-classification}, the actual $Ph$ is such
a constant sum. The full split and the original remainder theorem give
\[
 Q_{g_D}(h)=Q_{g_D}(Ph)+Q_{g_D}(h-Ph)\geq0.
\]
If the sum vanishes, both terms vanish. The preceding strict coefficient
makes $Ph=0$, while Theorem~\ref{thm:nonzonal-positivity-rigidity}
makes $h-Ph=0$. Thus $h=0$. This argument also covers forms whose
Haar average was zero from the outset. Conversely the zero section
has zero original action.

For the curvature formulation, evaluating the nonempty sphere gives
$\Lambda\geq\kappa>0$. Lemma~\ref{lem:curvature-profile-box} supplies
$m=1/\Lambda$, $M=1/\kappa$, and $M/m=\Lambda/\kappa$.
Apply the profile conclusion to these actual bounds.
\end{proof}

\begin{corollary}\label{cor:fixed-unit-probe-positive}
Under the profile bounds of Theorem~\ref{thm:produced-ck-safety},
$Q_{g_D}(M_D(1))>0$, including at the critical cap.
\end{corollary}

\begin{proof}
The unit meridional section is nonzero and CK by
Theorem~\ref{thm:invariant-ck-classification}. Nonnegativity and the zero
classification in Theorem~\ref{thm:produced-ck-safety} imply strict positivity.
\end{proof}

\section{Smooth sharpness and fixed-probe recovery}

\begin{lemma}\label{lem:even-smooth-box-approximation}
Let $a:\R\to\R$ be uniformly continuous and even, with
$\ell\leq a\leq u$ everywhere. There exist globally smooth even functions
$a_n:\R\to\R$ with the same bounds $\ell\leq a_n\leq u$ and
$a_n\to a$ uniformly on $\R$. The endpoints $\ell,u$ may coincide and need
not be positive.
\end{lemma}

\begin{proof}
Choose nonnegative normalized smooth bump kernels $\rho_n$, supported within
distance $r_n=1/(n+1)$ of zero, with $\int\rho_n=1$. Put
$b_n=\rho_n*a$. Each convolution is smooth: locally in its argument, every
derivative of the compactly supported kernel is integrated against the
continuous, hence locally integrable, function $a$ on a fixed compact set.
The nonnegative mass-one kernel makes $b_n(x)$ an average of numbers in the
closed convex interval $[\ell,u]$, so that interval is preserved exactly.
Uniform continuity and support shrinking to zero give
\[
 |b_n(x)-a(x)|
 \leq\sup_{|y|\leq r_n}|a(x-y)-a(x)|\longrightarrow0
\]
uniformly in $x$. Finally set $a_n(x)=(b_n(x)+b_n(-x))/2$.
This preserves smoothness and the interval, gives evenness, and still
converges uniformly because $a(-x)=a(x)$.
\end{proof}

For a continuous profile positive on $[-1,1]$, write
\[
 T(a)=\int_{-1}^1\frac{f_a(v)}{a(v)}\,dv,\qquad
 f_a(v)=2\int_v^1s a(s)\,ds .
\]
This is a scalar integral, not a smooth-metric construction for arbitrary
continuous $a$.

\begin{lemma}\label{lem:constant-probe-continuity}
Suppose $a,b$ are continuous on $[-1,1]$ and belong to a common positive box
$0<\ell\leq a,b\leq u$. If $|a-b|\leq\varepsilon$ there, then
\begin{equation}\label{eq:warp-ratio-lipschitz}
 |T(a)-T(b)|\leq
 8\left(\frac1\ell+\frac{u}{\ell^2}\right)\varepsilon .
\end{equation}
In particular, uniform convergence $a_n\to a$ on that interval with the same
positive box and continuous $a_n$ implies $T(a_n)\to T(a)$.
No derivative convergence or balance is required.
\end{lemma}

\begin{proof}
The primitives defining $f_a,f_b$ are continuous, and positive denominators
make their quotients continuous and integrable on the closed interval.
For $v\in[-1,1]$ the elementary integral estimates give
$|f_b(v)|\leq4u$ and $|f_a(v)-f_b(v)|\leq4\varepsilon$.
Here $\varepsilon\geq0$ follows from the assumed error bound at any point.
The exact quotient identity
\[
 \frac{f_a}{a}-\frac{f_b}{b}
 =\frac{f_a-f_b}{a}+\frac{f_b(b-a)}{ab}
\]
therefore bounds the pointwise absolute difference by
$4\varepsilon/\ell+4u\varepsilon/\ell^2$.
Integrating on an interval of length two proves
\eqref{eq:warp-ratio-lipschitz}.

For the limit assertion, uniform convergence gives continuity of $a$ and
passes the closed box to $a$, so the quantitative bound applies.
There is also a direct dominated-convergence proof: first pass to the limit
in the defining warp integrals, dominated by $u$, then in their quotients,
dominated by $4u/\ell$. These bounds establish actual integrability before
either passage to the limit.
\end{proof}

\begin{lemma}\label{lem:continuous-reciprocal-limit}
For $C\geq1$, let $p(t)=\max(0,\min(1,t))$ and define the actual extended
reciprocal optimizer by
\begin{equation}\label{eq:even-reciprocal-limit}
 a_{0,C}(v)=\exp\bigl(-k_{*,C}(p(|v|))\bigr).
\end{equation}
It is uniformly continuous and even on $\R$. Its bounds and physical formula are
\[
 \begin{aligned}
 \frac1C&\leq a_{0,C}(v)\leq1&&(v\in\R),\\
 a_{0,C}(v)&=\exp(-k_{*,C}(|v|))&&(-1\leq v\leq1).
 \end{aligned}
\]
It is balanced and satisfies
\begin{equation}\label{eq:reciprocal-limit-integral}
 T(a_{0,C})=2m(C),\qquad
 2\pi\left(2T(a_{0,C})-\frac43\right)
       =8\pi\left(m(C)-\frac13\right).
\end{equation}
\end{lemma}

\begin{proof}
The interval projection $p$ and the absolute-value map are Lipschitz.
The continuous function $t\mapsto\exp(-k_{*,C}(t))$ on the compact
interval $[0,1]$ is uniformly continuous. Their composition has the asserted
regularity; its formula is even and the optimizer's obstacle bounds give
the exact reciprocal box. On the physical interval the projection is the
identity. Beyond its endpoints it gives a constant extension.
The function $v a_{0,C}(v)$ is odd and continuous, so its integral on
$[-1,1]$ is zero.

For any globally continuous, physically positive, balanced $a$, the proof of
Lemma~\ref{lem:hemisphere-pair-substitution} uses no derivatives or smooth metric.
In particular the reflected-tail law follows from balance, all warp quotients
are integrable, and
\[
 T(a)=\I\!\left(\log\frac{M}{a(v)}\right)
      +\I\!\left(\log\frac{M}{a(-v)}\right)\qquad(M>0).
\]
Apply this with $a=a_{0,C}$ and $M=1$. On $[0,1]$, its northern logarithm
is exactly $k_{*,C}$; evenness identifies the southern logarithm with it.
The functional depends only on this interval, proving $T(a_{0,C})=2m(C)$.
The second identity in \eqref{eq:reciprocal-limit-integral} is algebra.
No smooth metric is assigned to the nonsmooth limiting optimizer.
\end{proof}

\begin{theorem}\label{thm:smooth-unit-probe-recovery}
For every $C\geq1$ there are globally smooth even profiles $a_n$ and
genuine produced pole data $D_n$ such that
\[
 \begin{gathered}
 D_n.a=a_n,\qquad
 \frac1C\leq a_n(v)\leq1\quad(v\in\R),\qquad
 \int_{-1}^1v a_n(v)\,dv=0,\\
 a_n\longrightarrow a_{0,C}\ \hbox{uniformly on }\R .
 \end{gathered}
\]
For their original metrics and the fixed scalar probe $r=1$,
\begin{equation}\label{eq:smooth-unit-probe-action-limit}
 Q_{g_{D_n}}(M_{D_n}(1))
       \longrightarrow8\pi\left(m(C)-\frac13\right).
\end{equation}
\end{theorem}

\begin{proof}
Apply Lemma~\ref{lem:even-smooth-box-approximation} to
$a_{0,C}$ with box $[1/C,1]$. Each approximant is even, so the exact
balance integral vanishes by oddness. Its positive lower bound permits the
existing smooth pole-factor and metric construction, yielding $D_n$ with
the explicit equation $D_n.a=a_n$.

For each of these actual smooth metrics, the accepted meridional reduction
and constant-probe calculation give
\[
 Q_{g_{D_n}}(M_{D_n}(1))
       =2\pi\left(2T(a_n)-\frac43\right).
\]
Lemma~\ref{lem:constant-probe-continuity} passes to the limit using the
common positive box. Equation~\eqref{eq:reciprocal-limit-integral} identifies
that limit and proves \eqref{eq:smooth-unit-probe-action-limit}.
Only the scalar probe is kept fixed; the actual metric and its canonical
meridional section are both produced from $a_n$.
\end{proof}

\begin{theorem}\label{thm:smooth-supercritical-ck-witness}
For every prescribed $C>C_{\mathrm{CK}}$, set
$C_0=(C+C_{\mathrm{CK}})/2$. There are a globally smooth even balanced
profile $a$ and actual produced data $D$ with $D.a=a$,
$1/C_0\leq a\leq1$ on $\R$, and
\[
 1\leq K_{g_D}\leq C_0<C
\]
on the sphere. The original section $M_D(1)$ is nonzero, is CK, and has
$Q_{g_D}(M_D(1))<0$.
\end{theorem}

\begin{proof}
The chosen cap satisfies $1<C_{\mathrm{CK}}<C_0<C$.
Theorem~\ref{thm:critical-pair-cap} gives $m(C_0)<1/3$.
Apply Theorem~\ref{thm:smooth-unit-probe-recovery} at $C_0$.
Its limit is negative, so an actual smooth member has negative action.
Retain that member's profile and producer equation $D.a=a$.
Lemma~\ref{lem:curvature-profile-box} converts its exact reciprocal box
to the displayed curvature bounds, including both poles.
The actual unit meridional section is nonzero and CK by
Theorem~\ref{thm:invariant-ck-classification}.
\end{proof}

\begin{corollary}\label{cor:critical-fixed-probe-recovery}
There is a sequence as in Theorem~\ref{thm:smooth-unit-probe-recovery} at
$C=C_{\mathrm{CK}}$ with $1\leq K_{g_{D_n}}\leq C_{\mathrm{CK}}$,
each $M_{D_n}(1)$ nonzero and CK, and
\[
 0<Q_{g_{D_n}}(M_{D_n}(1))\longrightarrow0.
\]
The profiles converge uniformly to the same $a_{0,C_{\mathrm{CK}}}$ and
retain their exact box, evenness, balance and producer equations.
\end{corollary}

\begin{proof}
The critical value is $m(C_{\mathrm{CK}})=1/3$, so the action limit in
\eqref{eq:smooth-unit-probe-action-limit} is zero.
Each member's curvature bounds follow from its reciprocal box, and its
nonzero CK property follows from the actual constant-probe theorem.
Corollary~\ref{cor:fixed-unit-probe-positive} proves that every smooth
action is strictly positive. No scaling of the scalar probe is used.
\end{proof}

\begin{theorem}\label{thm:geometric-hemisphere-stability}
For an original produced metric with $0<m\leq a\leq M$ on $[-1,1]$, set
$C=M/m$, $k(v)=\log(M/a(v))$ and $S=c^2+d^2>0$. Define
\begin{equation}\label{eq:geometric-stability-budget}
 B=\frac{Q_{g_D}(M_D(c)+Z_D(d))}{4\pi S}+\frac23-2m(C).
\end{equation}
Then $B=\I(k(v))+\I(k(-v))-2m(C)$ exactly, and
\begin{align*}
 \int_{-1}^1\bigl(k(v)-k_{*,C}(|v|)\bigr)^2\,dv&\leq B/c(C),\\
 \int_{-1}^1\bigl(k(v)-k(-v)\bigr)^2\,dv&\leq4B/c(C),\\
 \int_{-1}^1\left(k(v)-\frac{k(v)+k(-v)}2\right)^2\,dv&\leq B/c(C).
\end{align*}
Here $c(C)>0$ is the explicit stability constant from
Theorem~\ref{thm:stability}. If $S=0$, then $c=d=0$, the actual zonal
section and its action are zero; no normalized profile conclusion is asserted.
\end{theorem}

\begin{proof}
The profile bounds imply $M>0$, $C\geq1$, and continuity and the exact
obstacle bounds $0\leq k\leq\log C$ on the full interval.
Divide \eqref{eq:constant-zonal-pair-coefficient} by the nonzero quantity
$4\pi S$ and rearrange to obtain the stated identity for $B$.
Apply all three estimates of Corollary~\ref{cor:hemispheres} to this
literal full logarithm. This preserves their factor four for reflection
error and requires no equatorial symmetry.
For $S=0$, nonnegativity of squares forces both coefficients to vanish.
The actual section is then zero and the exact constant-zonal formula gives
zero action, without any division by $S$.
\end{proof}

\section{The represented metric class and the sharp CK threshold}

Let $\mathcal G_{\mathrm{rot}}$ be the class of actual smooth metrics $g$ on
$S^2$ for which there are produced data $D$ and a smooth diffeomorphism
$F:S^2\to S^2$ with $g=F^*g_D$. All tensor pullbacks below use the actual
derivative of this same map. No orientation condition is imposed on $F$.
This definition does not assume a representation theorem for arbitrary
abstract effective circle actions.

\begin{lemma}\label{lem:represented-metric-naturality}
Every produced metric belongs to $\mathcal G_{\mathrm{rot}}$, and this
class is closed under further smooth diffeomorphic pullback.
For a fixed representation $g=F^*g_D$, put $\Omega_F=F^*\Omega_D$. Then
\begin{equation}\label{eq:represented-curvature-volume}
 K_g(x)=K_{g_D}(F(x)),\qquad
 F_*\mu_g=\mu_{g_D},\qquad
 |\Omega_F|_g^2=2,\qquad \nabla^g\Omega_F=0 .
\end{equation}
The same positive curvature box holds for $g$ and $g_D$. The transported
curl satisfies
\[
 \operatorname{curl}_{g,\Omega_F}(F^*h)
 =\operatorname{curl}_{g_D,\Omega_D}(h)\circ F.
\]
\end{lemma}

\begin{proof}
The identity diffeomorphism represents $g_D$. If $g=G^*g_D$, then
$F^*g=(G\circ F)^*g_D$, proving closure with an actual representing map.
The actual derivative pullback has the two inverse laws
$(F^{-1})^*F^*=\operatorname{id}$ and
$F^*(F^{-1})^*=\operatorname{id}$ for tensors and metrics.
The curvature naturality and metric-volume change of variables used in
Lemma~\ref{lem:pullback-dissipation} give the first two identities in
\eqref{eq:represented-curvature-volume}. Surjectivity of $F$ gives both
directions of the curvature-box equivalence.

The area tensor evaluates by
$\Omega_F(x)(v,w)=\Omega_D(F(x))(dF_xv,dF_xw)$.
Actual tensor-norm and covariant-derivative naturality therefore transport
its squared norm and parallelness. Contracting the actual first derivative
against this same transported area gives the stated curl law.
For an orientation-reversing map we still use $\Omega_F$, rather than
asserting that a separately fixed area orientation was preserved.
Metric volume is the positive Riemannian measure, so its change of variables
requires no orientation-preserving premise.
\end{proof}

Fix $g=F^*g_D$. Write $\rho$ for the native circle rotations and define
\begin{equation}\label{eq:conjugated-action-projector}
 \rho_F(z)=F^{-1}\circ\rho(z)\circ F,\qquad
 P_F A=F^*P\bigl((F^{-1})^*A\bigr).
\end{equation}
The second expression is a smooth covariant tensor of the same rank as $A$.

\begin{theorem}\label{thm:conjugated-haar-projector}
The maps $\rho_F$ form a genuine circle action by isometries of $g$.
Its angle parametrization is jointly smooth on $\R\times S^2$ and has period
$2\pi$. For every smooth covariant tensor $A$ of rank $s$, including $s=0$,
every $x\in S^2$ and tuple $v_1,\ldots,v_s\in T_xS^2$,
\begin{equation}\label{eq:conjugated-haar-integral}
 (P_FA)_x(v_1,\ldots,v_s)=\frac1{2\pi}\int_0^{2\pi}
 A_{\rho_F(e^{i\theta})x}
 \bigl(d\rho_F(e^{i\theta})_xv_1,\ldots,
       d\rho_F(e^{i\theta})_xv_s\bigr)\,d\theta .
\end{equation}
The integrand is genuinely integrable. The projector is real-linear,
idempotent and invariant under $\rho_F$, and
$P_FA=A$ if and only if $A$ is invariant under every element of $\rho_F$.
It intertwines the original projector by $P_F(F^*A)=F^*(PA)$.
\end{theorem}

\begin{proof}
The identity, multiplication and inverse laws follow by canceling the
adjacent $F\circ F^{-1}$ in the explicit compositions.
The same cancellation and $\rho(z)^*g_D=g_D$ give
$\rho_F(z)^*g=g$. Joint smoothness follows by composing the jointly smooth
original angle action with $(\theta,x)\mapsto(\theta,F(x))$ and $F^{-1}$;
its period is unchanged. The actual tangent maps consequently preserve
the inner product of this same pulled-back metric.

Tensor pullback reverses composition, so for every tensor rank
\[
 \rho_F(z)^*A=F^*\rho(z)^*(F^{-1})^*A.
\]
Evaluate the definition of $P_FA$ at $x$ on the prescribed slot tuple,
then apply the original integral formula \eqref{eq:haar-covector-average}
at $F(x)$ on $(dF_xv_i)_i$. The displayed pullback identity identifies
each integrand with the derivative of the actual map $\rho_F(z)$, giving
\eqref{eq:conjugated-haar-integral}. Joint smoothness of the tensor orbit,
proved from the genuine family derivative, gives continuity at each fixed
fiber and slot tuple, hence integrability on the compact angle interval.

Linearity and the two inverse pullback laws give the asserted intertwining
and idempotence. Pulling the original invariant average through $F$ gives
$\rho_F(z)^*P_FA=P_FA$. Conversely invariance of $A$ becomes invariance
of $(F^{-1})^*A$ under $\rho$; the original fixed-point theorem then gives
$P_FA=A$. All these equations retain the same map and original tensor.
\end{proof}

\begin{theorem}\label{thm:represented-haar-rigidity}
For a fixed representation $g=F^*g_D$, $P_F$ commutes with the actual
first and second covariant derivatives, rough Laplacian and Ahlfors projection
in their respective tensor ranks. It is self-adjoint for the complete
four-term pairing of $g$. For every original smooth one-form $\eta$,
\begin{equation}\label{eq:represented-haar-split}
 Q_g(\eta)=Q_g(P_F\eta)+Q_g(\eta-P_F\eta),\qquad
 Q_g(\eta-P_F\eta)\geq0,
\end{equation}
and $Q_g(\eta-P_F\eta)=0$ if and only if $\eta-P_F\eta=0$.
Thus $Q_g(P_F\eta)\leq Q_g(\eta)$, with equality precisely for forms
invariant under $\rho_F$. No pinching assumption is required.
For every $g\in\mathcal G_{\mathrm{rot}}$, one actual representation supplies
these conclusions simultaneously for every $\eta$.
\end{theorem}

\begin{proof}
Put $h=(F^{-1})^*\eta$, so $\eta=F^*h$ by the actual inverse law.
The canonical naturality from Lemma~\ref{lem:pullback-dissipation},
followed by Theorem~\ref{thm:haar-derivatives}, gives, for example,
\[
 \nabla^g(P_F A)
 =F^*\nabla^{g_D}P((F^{-1})^*A)
 =F^*P\nabla^{g_D}((F^{-1})^*A)
 =P_F(\nabla^g A).
\]
Iterating this equality gives the actual second derivative. The accepted
rough-trace and Ahlfors naturality give the other commutations with all
covariant slots retained.

Complete-action and complete-pairing naturality transport the self-adjoint
identity and full Haar split from $g_D$. In particular
\[
 (F^{-1})^*(\eta-P_F\eta)=h-Ph,\qquad
 Q_g(\eta-P_F\eta)=Q_{g_D}(h-Ph).
\]
Theorem~\ref{thm:nonzonal-positivity-rigidity} supplies nonnegativity and
zero exactly when $h-Ph=0$. Injectivity of derivative pullback makes this
equivalent to the original remainder being zero. The split then gives the
comparison and its fixed-point equality, and
Theorem~\ref{thm:conjugated-haar-projector} identifies those fixed points.
Choosing a representation from the definition of the class, before choosing
$\eta$, proves the final full-class assertion.
\end{proof}

\begin{corollary}\label{cor:represented-zonal-geometry}
Fix any representation $g=F^*g_D$ and put
$M_F(r)=F^*M_D(r)$, $Z_F(s)=F^*Z_D(s)$. For globally smooth real probes,
\[
 Q_g(M_F(r)+Z_F(s))=2\pi(J_a[r]+J_a[s]).
\]
For every original smooth $\eta$, there are globally smooth $r,s$ with
$P_F\eta=M_F(r)+Z_F(s)$ and
\[
 Q_g(\eta)=2\pi(J_a[r]+J_a[s])+Q_g(\eta-P_F\eta)
          \geq2\pi(J_a[r]+J_a[s]).
\]
For these same probes, equality holds if and only if
$\eta=M_F(r)+Z_F(s)$. These statements hold for every represented metric,
with its actual representation retained.
\end{corollary}

\begin{proof}
Apply complete-action naturality to the actual sum $M_D(r)+Z_D(s)$
and use Theorem~\ref{thm:zonal-reduction}.
Classify $P((F^{-1})^*\eta)$ by
Theorem~\ref{thm:invariant-one-form-classification}, then pull that exact
section equality through $F$. This gives the globally smooth probes and
their stated original-section equation, including both transported poles.
The full split and zero-remainder theorem prove the remaining conclusions,
always using those same probes. No uniqueness of their extensions outside
the physical interval is claimed.
\end{proof}

\begin{theorem}\label{thm:represented-ck-safety}
Let $g\in\mathcal G_{\mathrm{rot}}$ and let $\eta$ be any original smooth
CK one-form. If $\kappa>0$, $\kappa\leq K_g\leq\Lambda$ everywhere and
$\Lambda/\kappa\leq C_{\mathrm{CK}}$, then
\[
 Q_g(\eta)\geq0,\qquad Q_g(\eta)=0\ \Longleftrightarrow\ \eta=0.
\]
In particular every nonzero such form has strictly positive action, including
at the critical cap. For any representing $D,F$, the actual $P_F$ preserves
the CK equation, and $P_F\eta=M_F(c)+Z_F(d)$ for some real constants $c,d$.
\end{theorem}

\begin{proof}
Choose $g=F^*g_D$ and put $h=(F^{-1})^*\eta$.
The actual derivative/Ahlfors pullback law makes $h$ CK for $g_D$.
Lemma~\ref{lem:represented-metric-naturality} supplies the same curvature
box on $g_D$, so Theorem~\ref{thm:produced-ck-safety} applies.
Complete-action naturality gives $Q_g(\eta)=Q_{g_D}(h)$, and inverse
pullback makes $h=0$ equivalent to $\eta=0$. This proves the full
arbitrary-form conclusion, with no nonzero-average hypothesis.

The Ahlfors-derivative commutation in
Theorem~\ref{thm:represented-haar-rigidity} sends its zero equation to the
zero equation of $P_F\eta$. Alternatively transport the original CK Haar
preservation theorem. The original constant-probe classification of $Ph$,
pulled back through $F$, gives the last section equality.
\end{proof}

\begin{theorem}\label{thm:represented-geometric-stability}
Fix $g=F^*g_D$, use its profile height $z_F=z\circ F$, and suppose
$0<m\leq a\leq M$ on $[-1,1]$. Set $C=M/m$, $k(v)=\log(M/a(v))$ and
$S=c^2+d^2>0$. Then
\begin{equation}\label{eq:represented-stability-budget}
 B_F=\frac{Q_g(M_F(c)+Z_F(d))}{4\pi S}+\frac23-2m(C)
     =\I(k(v))+\I(k(-v))-2m(C),
\end{equation}
and all three inequalities in Theorem~\ref{thm:geometric-hemisphere-stability}
hold with $B_F$ in place of $B$. The reflection factor is four.
For $S=0$, both coefficients, the pulled-back section and its action vanish,
without a normalized profile conclusion.
Every represented metric has one representation supporting these conclusions
for all profile bounds and coefficient pairs simultaneously.
\end{theorem}

\begin{proof}
The section is literally $F^*(M_D(c)+Z_D(d))$. Pullback naturality identifies
its complete action with the original constant-zonal action on $g_D$.
Substitute that equality into the exact budget identity and all three
inequalities of Theorem~\ref{thm:geometric-hemisphere-stability}.
The profile logarithm is the same generating function in height parameter
$v$; on the represented sphere it is evaluated at $z_F$.
For zero coefficient sum use the original zero-coefficient law and the
linearity and zero preservation of derivative pullback.
Fixing the representation before the parameters proves the final assertion.
\end{proof}

Define $\mathcal S_{\mathrm{CK}}$ to consist of the real numbers $C\geq1$
such that $Q_g(\eta)\geq0$ for every $g\in\mathcal G_{\mathrm{rot}}$,
every actual curvature box $\kappa>0$, $\kappa\leq K_g\leq\Lambda$ with
$\Lambda/\kappa\leq C$, and every original smooth CK one-form $\eta$.
Define the universal CK threshold as $\sup\mathcal S_{\mathrm{CK}}$.

\begin{theorem}\label{thm:universal-ck-threshold}
The actual universal safe-cap set is
\begin{equation}\label{eq:universal-ck-safe-set}
 \mathcal S_{\mathrm{CK}}=[1,C_{\mathrm{CK}}],
 \qquad
 \sup\mathcal S_{\mathrm{CK}}
 =\left(\frac{5+\sqrt{13}}3\right)^{1/3}
       \exp\left(\frac{4+2\sqrt{13}}9\right).
\end{equation}
It is nonempty and bounded above. Every smooth nonzero CK form in a
represented critical safe box has positive action, while the same fixed
unit probes of Corollary~\ref{cor:critical-fixed-probe-recovery} lie in this
class and have positive actions tending to zero. The supercritical witnesses
of Theorem~\ref{thm:smooth-supercritical-ck-witness} also lie in this class
with their exact producer and curvature equations.
\end{theorem}

\begin{proof}
If $1\leq C\leq C_{\mathrm{CK}}$, every curvature-box ratio at most $C$
is at most $C_{\mathrm{CK}}$. Theorem~\ref{thm:represented-ck-safety}
therefore proves $C\in\mathcal S_{\mathrm{CK}}$ with all the stated
universal quantifiers.
Conversely let $C>C_{\mathrm{CK}}$ and choose exactly
$C_0=(C+C_{\mathrm{CK}})/2$. The already constructed smooth witness has
$1\leq K_{g_D}\leq C_0<C$, a nonzero CK unit meridional section, and negative
original action. Its metric belongs to $\mathcal G_{\mathrm{rot}}$ through
the identity diffeomorphism. Applying the proposed safety property with
$\kappa=1$ and $\Lambda=C_0$ is a contradiction.
Caps below one are excluded by definition, proving the set equality.

Since $C_{\mathrm{CK}}>1$, the interval is nonempty and bounded above by its
included right endpoint. The real supremum laws and
\eqref{eq:critical-pair-cap} give the second equality in
\eqref{eq:universal-ck-safe-set}.
The universal strict positivity follows from
Theorem~\ref{thm:represented-ck-safety}. For recovery, retain exactly the
profiles, produced metrics and fixed probes from the earlier critical
sequence and add their identity representations. This neither constructs
a second sequence nor changes their action limit. The same identity
embedding retains the original supercritical witness equations.
\end{proof}

\section{Normalized area geometry and complete action realization}

\begin{lemma}\label{lem:area-second-primitive}
For a continuous real function $u$ on $\R$ and a fixed $a\in\R$, put
$S(x)=\int_a^x(x-t)u(t)\,dt$. Then
\[
 S(x)=x\int_a^x u(t)\,dt-\int_a^x t u(t)\,dt,\qquad
 S'(x)=\int_a^x u(t)\,dt .
\]
If $u$ is globally smooth, both its first primitive and $S$ are globally smooth.
\end{lemma}

\begin{proof}
All integrands are continuous on every compact integration interval, so
linearity gives the first identity for either orientation of the interval.
Differentiate its two ordinary primitives using the fundamental theorem of
calculus. The two terms $xu(x)$ cancel, leaving the stated derivative.
For smooth $u$, the first primitive is differentiable with smooth derivative
$u$, and $S$ is differentiable with that smooth first primitive as derivative.
Induction on the derivative order proves global smoothness.
\end{proof}

For a curvature coefficient $K:\R\to\R$, define the actual normalized primitives
\begin{equation}\label{eq:normalized-area-primitives}
 V_K(x)=-1+\int_0^xK(t)\,dt,\qquad
 f_K(x)=2x-2\int_0^x(x-t)K(t)\,dt,\qquad
 f_{1,K}(x)=2-2\int_0^xK(t)\,dt .
\end{equation}

\begin{theorem}\label{thm:positive-normalized-area-geometry}
Suppose $K$ is continuous on $\R$, strictly positive on $[0,1]$, and satisfies
the two exact normalized moments
\[
 \int_0^1K(x)\,dx=2,\qquad \int_0^1xK(x)\,dx=1.
\]
The actual derivative identities are
\[
 V_K'=K,\qquad f_K'=f_{1,K}=-2V_K,\qquad
 f_{1,K}'=f_K''=-2K .
\]
They hold on $\R$. Moreover,
\[
 V_K(0)=-1,\quad V_K(1)=1,\quad
 f_K(0)=f_K(1)=0,\quad f_{1,K}(0)=2,\quad f_{1,K}(1)=-2.
\]
The map $V_K$ is a strictly increasing bijection from $[0,1]$ onto $[-1,1]$,
and $\int_0^1V_K=0$. The warp $f_K$ is strictly concave on $[0,1]$ and
strictly positive on $(0,1)$. If $K$ is globally smooth, all three primitives
are globally smooth.
\end{theorem}

\begin{proof}
Apply Lemma~\ref{lem:area-second-primitive} and the fundamental theorem to
the literal definitions \eqref{eq:normalized-area-primitives}. These also
give $f_{1,K}=-2V_K$ by algebra. The left endpoints follow directly.
The zeroth moment gives the right height and slope, while
\[
 f_K(1)=2-2\left(\int_0^1K(t)\,dt-\int_0^1tK(t)\,dt\right)=0 .
\]
Positive $V_K'=K$ gives strict monotonicity. Continuity, the endpoint values
and the intermediate value theorem give the asserted bijection.
Integrating $V_K=-f_K'/2$ between the two zero warp endpoints gives its
zero integral, with integrability supplied by continuity.

The actual second derivative is strictly negative on $[0,1]$, so $f_K$
is strictly concave there. For $0<x<1$, strict concavity between its two
zero endpoint values gives $f_K(x)>0$. This step uses positive curvature;
the two moment equations alone do not imply it.
Smoothness follows from the last part of Lemma~\ref{lem:area-second-primitive}.
\end{proof}

\begin{lemma}\label{lem:smooth-closed-interval-inverse}
Let $a<b$ and let $V:\R\to\R$ be globally smooth with $V'>0$ on $[a,b]$.
There exists a globally smooth real function $X$ such that
\[
 X([V(a),V(b)])\subset[a,b],\qquad
 X(V(x))=x\ (a\leq x\leq b),\qquad
 V(X(v))=v\ (V(a)\leq v\leq V(b)).
\]
It has $X(V(a))=a$, $X(V(b))=b$, and
$X'(v)V'(X(v))=1$ on the entire closed target interval, including both endpoints.
The conclusion does not require $X$ to invert $V$ outside those intervals.
\end{lemma}

\begin{proof}
Strict monotonicity and the intermediate value theorem produce the actual
closed-interval bijection. Its continuous inverse follows from compactness
of the source and the Hausdorff property of the target. Extend this inverse
continuously to the line by clamping its input to the target interval.

At a target point $v$, including an endpoint, put $x=V^{-1}(v)$.
Since $V'(x)>0$, the local smooth inverse theorem gives a smooth inverse
branch near $v$. Continuity of the clamped inverse and the original inverse
equation show that it agrees with this branch within the target interval
near $v$. Thus the actual inverse is smooth on that closed interval,
with its one-sided endpoint jets.

A smooth function on a nondegenerate closed real interval has a global
smooth extension agreeing on the interval. One construction realizes each
endpoint jet $(c_j)$ by a cutoff series
$\sum_j c_j(t-t_0)^j\chi((t-t_0)/\varepsilon_j)/j!$,
where $\chi$ is smooth, compactly supported and equal to one near zero.
Choose $\varepsilon_j>0$ small enough that the $C^k$ norm of its $j$th term
is at most $2^{-j}$ for every $k<j$. Every derivative series then converges
uniformly after finitely many initial terms and has exactly the prescribed
jet at $t_0$. Glue these left and right realizations to the given function;
matching all one-sided derivatives makes the result globally smooth.
Apply this extension construction to the actual interval inverse.

The extension retains both inverse equations and endpoints. Differentiate
$V(X(v))=v$ within the nondegenerate closed target interval. Uniqueness of
the derivative within that interval, also at its endpoints, gives
$V'(X(v))X'(v)=1$. This identifies the ambient derivative of the global
extension without imposing any exterior inverse law.
\end{proof}

For actual primitives $f=f_K$, $f_1=f_{1,K}$ and a smooth real probe $r$, define
the complete normalized area action by
\begin{equation}\label{eq:complete-area-action}
 A_K(r)=\int_0^1
 \left[
 f(fr''+2f_1r'-Kr)^2
 -Kf_1r\left(fr'+\frac{f_1r}{2}\right)
 +K^2fr^2
 \right]dx .
\end{equation}
For continuous $K$ and smooth $r$, every factor in this density is continuous,
so the integral is genuine.

\begin{lemma}\label{lem:actual-area-probe-jets}
Let $K,r$ be globally smooth, with $K>0$ on $[0,1]$ and $\int_0^1K=2$.
Suppose $D$ is original produced data and $R$ is globally smooth, with the
physical equations
\[
 D.a(V_K(x))=1/K(x),\qquad R(V_K(x))=r(x)\qquad(0\leq x\leq1).
\]
Write $a=D.a$ and $H=\operatorname{warp}(a)/a$ for the original height-profile
quantities. On the entire closed area interval,
\begin{align*}
 \operatorname{warp}(a)(V_K(x))&=f_K(x),&
 a'(V_K(x))&=-K'(x)/K(x)^3,\\
 R'(V_K(x))&=r'(x)/K(x),&
 R''(V_K(x))&=r''(x)/K(x)^2-r'(x)K'(x)/K(x)^3,\\
 H(V_K(x))&=f_K(x)K(x),&
 H'(V_K(x))&=f_{1,K}(x)+f_K(x)K'(x)/K(x).
\end{align*}
Consequently the actual operator from \eqref{eq:meridional-operator} satisfies
\begin{equation}\label{eq:area-operator-chain}
 L_aR(V_K(x))
   =\frac{f_K(x)r''(x)+2f_{1,K}(x)r'(x)-K(x)r(x)}{K(x)} .
\end{equation}
\end{lemma}

\begin{proof}
The actual height derivative is $V_K'=K$.
The derivative of $\operatorname{warp}(a)(V_K(x))$ is
\[
 -2V_K(x)a(V_K(x))K(x)=-2V_K(x)=f_K'(x).
\]
Both functions vanish at $x=0$, using $V_K(0)=-1$ and the original
balance equation of $D$. Equality of these actual derivatives and continuity
give the warp identity on the full interval.

Differentiate the coefficient and probe equations within $[0,1]$.
Since both sides have actual ambient derivatives at every point, uniqueness
within the nondegenerate interval gives the displayed first derivatives,
including the endpoints. Differentiate the first probe derivative equation
once more to obtain the stated second derivative. Each division is justified
by $K(x)>0$.

The value of $H$ follows from the coefficient and warp equations.
Substitution in the accepted formula
$H'(v)=-2v-H(v)a'(v)/a(v)$ gives its displayed derivative, since
$f_{1,K}=-2V_K$. Finally substitute these actual jets into $L_aR$.
The two terms involving $f_Kr'K'/K^2$ cancel and the remaining terms give
\eqref{eq:area-operator-chain}. No independent first or second jet is supplied.
\end{proof}

\begin{theorem}\label{thm:complete-area-action-bridge}
Under the hypotheses of Lemma~\ref{lem:actual-area-probe-jets}, the original
smooth metric and section satisfy
\[
 Q_{g_D}(M_D(R))=2\pi A_K(r).
\]
The identity retains every term of the original complete action.
\end{theorem}

\begin{proof}
Let $j_a[R]$ denote the original meridional density in
\eqref{eq:meridional-action}. Multiply its value at $V_K(x)$ by the actual
Jacobian $V_K'(x)=K(x)$. Lemma~\ref{lem:actual-area-probe-jets} and
$-2V_K=f_{1,K}$ give the literal identity
\[
 j_a[R](V_K(x))K(x)
 =f_K(f_Kr''+2f_{1,K}r'-Kr)^2
  -Kf_{1,K}r\left(f_Kr'+\frac{f_{1,K}r}{2}\right)
  +K^2f_Kr^2 .
\]
It holds at both endpoints as well. In particular no endpoint term is discarded.
Positive curvature and the zeroth moment map $V_K([0,1])$ into $[-1,1]$,
with endpoints $-1,1$. The original density is continuous on that physical
height interval. The interval change-of-variables theorem therefore applies
with continuity only on the actual image; no exterior positivity of $D.a$
is needed. It identifies $\int_{-1}^1j_a[R]$ with
\eqref{eq:complete-area-action}. The original meridional reduction then
supplies the unchanged factor $2\pi$.
\end{proof}

\begin{theorem}\label{thm:fully-tied-area-realization}
Let $K,r:\R\to\R$ be globally smooth, suppose $K>0$ on $[0,1]$, and impose
the two moments $\int_0^1K=2$, $\int_0^1xK=1$.
There exist a globally smooth $X$, original produced data $D$, and a
globally smooth $R$ with
\[
 V_K(0)=-1,\quad V_K(1)=1,\quad X(-1)=0,\quad X(1)=1,
\]
both physical interval mapping laws, both inverse laws
$X(V_K(x))=x$ on $[0,1]$ and $V_K(X(v))=v$ on $[-1,1]$, and
$X'(v)K(X(v))=1$ on the closed height interval. They satisfy $R=r\circ X$
globally and, for every $x\in[0,1]$,
\begin{equation}\label{eq:area-realization-tying}
 D.a(V_K(x))=\frac1{K(x)},\qquad
 R(V_K(x))=r(x),\qquad
 \operatorname{warp}(D.a)(V_K(x))=f_K(x).
\end{equation}
All coefficient/probe derivative equations in
Lemma~\ref{lem:actual-area-probe-jets} hold, including the endpoints.
For every actual sphere point $p$,
\begin{equation}\label{eq:area-realization-curvature-action}
 K_{g_D}(p)=K(X(z(p))),\qquad
 Q_{g_D}(M_D(R))=2\pi A_K(r).
\end{equation}
\end{theorem}

\begin{proof}
Apply Lemma~\ref{lem:smooth-closed-interval-inverse} to the actual globally
smooth primitive $V_K$ on $[0,1]$. Its derivative is $K>0$ and its endpoints
are $-1,1$, so the resulting $X$ has all the stated coordinate laws.
The smooth coefficient $K\circ X$ is nonzero on $[-1,1]$.
The smooth quotient extension used in
Theorem~\ref{thm:invariant-one-form-classification} supplies a globally
smooth $a$ with $(K\circ X)a=1$ there. Thus $a>0$ on the physical interval
and $a(V_K(x))=1/K(x)$. Put $R=r\circ X$.

The actual change of variables and the physical reciprocal equation yield
\[
 \int_{-1}^1v a(v)\,dv
 =\int_0^1 V_K(x)a(V_K(x))K(x)\,dx
 =\int_0^1 V_K(x)\,dx=0 .
\]
Continuity establishes every integral and the last equality follows from
the two moments by Theorem~\ref{thm:positive-normalized-area-geometry}.
The existing smooth positive balanced-profile construction therefore produces
$D$ with the exact equation $D.a=a$.
Both input compatibility equations for
Lemma~\ref{lem:actual-area-probe-jets} have now been constructed, not assumed.
That lemma and Theorem~\ref{thm:complete-area-action-bridge} supply the warp,
actual derivative and original-action identities.

Finally $z(p)\in[-1,1]$, so $x=X(z(p))$ belongs to $[0,1]$ and
$V_K(x)=z(p)$. The original intrinsic curvature law gives
$K_{g_D}(p)=1/D.a(z(p))=K(x)$, including both poles.
The chosen maps, profile and probe remain tied in all these equations.
\end{proof}

\begin{lemma}\label{lem:reciprocal-area-coordinate}
Let $a:\R\to\R$ be continuous on $\R$ and strictly positive on $[-1,1]$. Set
\[
 L_a=\int_{-1}^1a(v)\,dv,\qquad
 X_a(v)=\frac1{L_a}\int_{-1}^v a(s)\,ds.
\]
Then $L_a>0$ and $X_a$ maps $[-1,1]$ continuously and strictly increasingly
onto $[0,1]$, with $X_a(-1)=0$, $X_a(1)=1$ and $X_a'=a/L_a$.
Let $V_a$ be the physical inverse, extended constantly to $-1$ and $1$
outside $[0,1]$, and put $K_a(x)=L_a/a(V_a(x))$.
The functions $V_a,K_a$ are continuous on $\R$, $K_a>0$, and
\[
 \begin{aligned}
 V_a(X_a(v))&=v&&(-1\le v\le1),\\
 X_a(V_a(x))&=x&&(0\le x\le1),\\
 K_a(X_a(v))&=\frac{L_a}{a(v)}&&(-1\le v\le1).
 \end{aligned}
\]
If $\int_{-1}^1va(v)\,dv=0$, then
\begin{equation}\label{eq:reciprocal-area-moments-warp}
 \begin{gathered}
 \int_0^1K_a=2,\qquad \int_0^1xK_a(x)\,dx=1,\\
 f_{K_a}(X_a(v))=\frac{\operatorname{warp}(a)(v)}{L_a},\qquad
 f_{K_a}'(X_a(v))=-2v\quad(-1\le v\le1).
 \end{gathered}
\end{equation}
Under the same balance hypothesis, the complete area action satisfies
\begin{equation}\label{eq:reciprocal-area-unit-action}
 A_{K_a}(1)=2\int_{-1}^1\frac{\operatorname{warp}(a)(v)}{a(v)}\,dv-\frac43.
\end{equation}
\end{lemma}

\begin{proof}
Positivity gives $L_a>0$ and $X_a'>0$ on the physical interval. Compactness
and strict monotonicity give the homeomorphism onto its endpoint interval;
clamping the inverse gives the stated continuous extension. In particular its
range remains in $[-1,1]$, so its curvature is continuous and positive on the
whole line. All inverse identities above concern the indicated physical
intervals.

The chain rule and the curvature tying equation give
$(V_{K_a}\circ X_a)'=K_a(X_a)a/L_a=1$.
Both sides agree at $v=-1$, hence $V_{K_a}(X_a(v))=v$.
Thus $f_{K_a}'(X_a(v))=-2v$ and the zeroth moment is two.
The derivative of $f_{K_a}\circ X_a$ is $-2va/L_a$, which equals the
derivative of $\operatorname{warp}(a)/L_a$.
Balance makes both functions zero at $-1$, proving the warp equation.
At $v=1$ the warp vanishes; the corrected triangular primitive, together with
the zeroth moment, then gives the first moment.

For the unit probe the full density reduces to
$2K_a^2f_{K_a}-K_a(f_{K_a}')^2/2$.
After multiplication by the actual Jacobian $X_a'=a/L_a$, it becomes
$2\operatorname{warp}(a)/a-2v^2$ on $[-1,1]$.
Continuity gives integrability, and ordinary interval substitution gives
\eqref{eq:reciprocal-area-unit-action}. This calculation requires no derivative
of $a$ or $K_a$ and assigns no smooth metric to these continuous data.
\end{proof}

\begin{theorem}\label{thm:critical-area-curvature}
Take $a_*=a_{0,C_{\mathrm{CK}}}$ from \eqref{eq:even-reciprocal-limit} and use the
preceding construction to define $L_*,X_*,V_*,K_*$.
Then $K_*$ has both normalized moments, is continuous and positive, and obeys
\[
 L_*\le K_*(x)\le L_*C_{\mathrm{CK}},\qquad
 L_*<2<L_*C_{\mathrm{CK}}.
\]
On $[0,1]$ it is symmetric about $1/2$, nonincreasing on $[0,1/2]$ and
nondecreasing on $[1/2,1]$.
There are $0<l<u<1$ such that $K_*=L_*$ on $[l,u]$,
and $f_{K_*}>0$ throughout $(0,1)$.
The actual complete unit-probe area action is zero:
\begin{equation}\label{eq:critical-area-zero-action}
 A_{K_*}(1)=0.
\end{equation}
\end{theorem}

\begin{proof}
The actual reciprocal optimizer is even, continuous, lies in $[1/C_{\mathrm{CK}},1]$,
equals one on its positive central contact interval and equals $1/C_{\mathrm{CK}}$
at the outer contacts. Its logarithmic profile is nondecreasing on the northern
half, so the reciprocal is nonincreasing there. The strict inequalities for
$L_*$ follow by integrating the two box bounds: each is strict on a nonempty
interval because $C_{\mathrm{CK}}>1$ and both contact regions are nonempty.
Evenness gives balance and $X_*(-v)=1-X_*(v)$, whence $X_*(0)=1/2$ and
$V_*(1-x)=-V_*(x)$ on $[0,1]$.
These identities prove the symmetry of $K_*$, and the increasing inverse
transports the two hemisphere monotonicity laws. A closed subinterval of the
central contact region maps to an interior constant-curvature plateau.
Positive curvature and the two moments give strict interior warping positivity
by Theorem~\ref{thm:positive-normalized-area-geometry}.
Finally the critical variational value is $m(C_{\mathrm{CK}})=1/3$,
and the reciprocal warp-ratio integral is twice that value. Substitution in
\eqref{eq:reciprocal-area-unit-action} gives $2(2/3)-4/3=0$.
\end{proof}

\begin{lemma}\label{lem:plateau-first-variation}
Let $K:\R\to\R$ be continuous, and fix $0<l<u<1$ with $K=\kappa$ on $(l,u)$.
Write $f=f_K$, and let $\chi$ be globally smooth with support compactly
contained in $(l,u)$. The complete action has the exact variation
\begin{equation}\label{eq:plateau-exact-variation}
 A_K(1-\varepsilon\chi)=A_K(1)
 -2\varepsilon\kappa^2\int_0^1 f\chi
 +\varepsilon^2 A_K(\chi).
\end{equation}
\end{lemma}

\begin{proof}
Expansion of the full six-entry density gives the coefficient of $-\varepsilon$
\[
 -2Kf^2\chi''-5Kff'\chi'+(4K^2f-K(f')^2)\chi.
\]
On the support of $\chi$, use $f''=-2\kappa$ to write this as
\[
 2\kappa^2 f\chi+
 \bigl(-2\kappa f^2\chi'-\kappa ff'\chi\bigr)'.
\]
Off that support, $\chi$ and both derivatives vanish, so the equality remains
valid. The flux vanishes at both endpoints. All terms are continuous, and
integrating the identity proves \eqref{eq:plateau-exact-variation}, with its
quadratic term equal to the actual action of the same bump.
\end{proof}

\begin{theorem}\label{thm:critical-plateau-instability}
The actual critical curvature $K_*$ admits a globally smooth probe $r$ which
equals one near both endpoints, is nonconstant on $[0,1]$, has
$r(0)=1$, $r'(0)=0$, and satisfies $A_{K_*}(r)<0$.
\end{theorem}

\begin{proof}
Choose a nonnegative, nonzero smooth bump $\chi$ supported strictly inside the
positive plateau. Since $f_{K_*}>0$ there, its weighted pairing is positive.
Set
\[
 A=L_*^2\int_0^1 f_{K_*}\chi>0,\qquad
 B=A_{K_*}(\chi),\qquad
 \varepsilon=\frac{A}{|B|+1}>0,\qquad r=1-\varepsilon\chi.
\]
Then $\varepsilon|B|<A$, and therefore
$-2\varepsilon A+\varepsilon^2B<0$.
Equations~\eqref{eq:critical-area-zero-action} and
\eqref{eq:plateau-exact-variation} give the negative action.
The same probe has the actual derivatives $r'=-\varepsilon\chi'$ and
$r''=-\varepsilon\chi''$. Its support condition gives constancy near both
endpoints and the stated initial data; positivity of the pairing ensures
physical nonconstancy. Only continuity of the curvature has been used;
smooth geometric recovery is a subsequent step.
\end{proof}

\section{Smooth recovery and strict threshold separation}

\begin{lemma}\label{lem:sampled-symmetric-bernstein}
For $n\geq1$ and a real function $q$, set
\[
 B_nq(y)=\sum_{j=0}^n\binom nj y^j(1-y)^{n-j}q(j/n).
\]
This is a polynomial on $\R$. On $[0,1]$ it preserves every closed box
containing the sampled values, and it is nondecreasing if $q$ is
nondecreasing there. If $q$ is bounded on $[0,1]$, then
$B_nq(y)\to q(y)$ at each point where the restriction of $q$ to $[0,1]$
is continuous, including either endpoint with relative continuity.

For a real function $K$, define
\[
 \rho(x)=(2x-1)^2,\qquad q_K(y)=K\left(\frac{1+\sqrt y}{2}\right),
 \qquad S_nK(x)=B_nq_K(\rho(x)).
\]
The function $S_nK$ is globally smooth and satisfies
$S_nK(1-x)=S_nK(x)$ for every real $x$.
If $K$ is nondecreasing on $[1/2,1]$, then $S_nK$ is nondecreasing there
and nonincreasing on $[0,1/2]$. A box for $K$ on $[0,1]$ is preserved by
$S_nK$ on $[0,1]$.
If also $K(1-x)=K(x)$ for $x\in[0,1]$, then
$q_K(\rho(x))=K(x)$ on that interval.
\end{lemma}

\begin{proof}
The Bernstein basis is nonnegative and sums to one on $[0,1]$, proving
box preservation. The exact adjacent-difference derivative formula is
\[
 (B_{n+1}q)'(y)=(n+1)\sum_{j=0}^{n}\binom nj y^j(1-y)^{n-j}
 \left(q\left(\frac{j+1}{n+1}\right)-q\left(\frac j{n+1}\right)\right).
\]
Thus monotone sampled values give a nonnegative derivative.
For convergence at a relative continuity point $y$, subtract $q(y)$
inside the convex sum. Split the nodes into those within distance $\delta$
of $y$ and their complement. The first part is bounded by the local
oscillation. If $|q(z)-q(y)|\leq D$ on the interval, the complementary
part is at most
\[
 \frac{D}{\delta^2}\sum_{j=0}^n
   (j/n-y)^2\binom nj y^j(1-y)^{n-j}
 =\frac{D\,y(1-y)}{n\delta^2}\longrightarrow0.
\]
This proves the asserted pointwise convergence without assuming global
continuity of $q$.

The radial argument lies in $[1/2,1]$ when $y\in[0,1]$.
Both it and $\rho$ on $[1/2,1]$ are nondecreasing, giving the right-hand
order law. Since $\rho(1-x)=\rho(x)$, reflection gives the other order law.
The polynomial composition gives global smoothness. Finally
$\sqrt{(2x-1)^2}=|2x-1|$; the radial argument is $x$ on the right half
and $1-x$ on the left, proving the reconstruction identity.
\end{proof}

\begin{theorem}\label{thm:symmetric-exact-moment-recovery}
Let $\ell<m<u$. Suppose $K:\R\to\R$ has a strongly measurable
restriction to $[0,1]$ and satisfies
\[
 \ell\leq K(x)\leq u,\quad K(1-x)=K(x)\quad(0\leq x\leq1),
 \qquad \int_0^1K=m,
\]
and is nondecreasing on $[1/2,1]$.
There are functions $G_n:\R\to\R$, each smooth on $\R$, such that for
every $n$ their physical restrictions are symmetric, nonincreasing on
$[0,1/2]$, nondecreasing on $[1/2,1]$, and obey
\begin{equation}\label{eq:exact-moment-recovery}
 \ell\leq G_n(x)\leq u\quad(0\leq x\leq1),\qquad
 \int_0^1G_n=m,\qquad \int_0^1xG_n(x)\,dx=\frac m2.
\end{equation}
Moreover $\int_0^1|G_n-K|\to0$, and there is a strictly increasing
subsequence index $\nu$ with $G_{\nu(n)}(x)\to K(x)$ for almost every
$x\in[0,1]$.
\end{theorem}

\begin{proof}
We first construct an approximant with any prescribed $L^1$ error
$\varepsilon>0$. Choose
\[
 0<\eta<\min\left(1,\frac{\varepsilon}{u-\ell}\right),\qquad
 K_\eta=(1-\eta)K+\eta m,\qquad
 \delta=\eta\min(m-\ell,u-m)>0.
\]
This preserves symmetry, hemisphere order and mean $m$, and gives the
strict box $\ell+\delta\leq K_\eta\leq u-\delta$ on $[0,1]$.
Also $\int_0^1|K_\eta-K|\leq\eta(u-\ell)<\varepsilon$.

Apply Lemma~\ref{lem:sampled-symmetric-bernstein} to $K_\eta$.
The radial profile $q_{K_\eta}$ is monotone on $[0,1]$, so its relative
discontinuity set is countable. The inverse image of this set under
$\rho$ is countable: $\rho$ is injective separately on each closed
hemisphere. Hence $S_nK_\eta\to K_\eta$ almost everywhere on $[0,1]$.
The unchanged strict box bounds the absolute difference by its width;
dominated convergence gives $L^1$ convergence and convergence of the means.

Put
\[
 c_n=m-\int_0^1S_nK_\eta,\qquad H_n=S_nK_\eta+c_n.
\]
Then $c_n\to0$ and $\int_0^1H_n=m$ exactly.
For large $n$, $|c_n|<\delta$, so $H_n$ stays in the original box
$[\ell,u]$ on the physical interval. Adding the constant changes neither
smoothness nor reflection nor either hemisphere order law.
Furthermore
\[
 \int_0^1|H_n-K|
 \leq\int_0^1|S_nK_\eta-K_\eta|+|c_n|+\eta(u-\ell).
\]
Choose the degree after fixing $\eta$ and $\delta$, making the right side
less than $\varepsilon$. Reflection gives the other exact moment:
substitution $x\mapsto1-x$ implies
$2\int_0^1xH_n(x)\,dx=\int_0^1H_n=m$.
All integral operations are justified by measurability and the fixed
finite physical box.

Choose such approximants for errors $1/(n+1)$. Their errors tend to zero,
proving the sequence assertion. Finally choose strictly increasing indices
whose $L^1$ errors have finite sum. Tonelli's theorem makes the sum of the
pointwise absolute errors finite almost everywhere, and therefore those
errors tend to zero. Every selected member retains \eqref{eq:exact-moment-recovery}.
\end{proof}

\begin{lemma}\label{lem:complete-area-action-convergence}
Let $K_n,K:\R\to\R$ be continuous on $\R$, and fix one globally smooth
$r:\R\to\R$. Suppose that $|K_n(x)|\leq B$ for every $n$ and
$x\in[0,1]$, that $\int_0^1|K_n-K|\to0$, and that $K_n\to K$ almost
everywhere on $[0,1]$. Then
\begin{equation}\label{eq:area-primitive-lone-bounds}
 \begin{aligned}
 |f_{K_n}(x)-f_K(x)|&\leq2\int_0^1|K_n-K|,\\
 |f_{1,K_n}(x)-f_{1,K}(x)|&\leq2\int_0^1|K_n-K|
 \qquad(0\leq x\leq1).
 \end{aligned}
\end{equation}
Thus both primitives converge uniformly on $[0,1]$, and the complete
actions of this same probe satisfy $A_{K_n}(r)\to A_K(r)$.
\end{lemma}

\begin{proof}
Subtract the actual primitive formulas
\eqref{eq:normalized-area-primitives}. On $0\leq s\leq x\leq1$,
the kernel $|x-s|$ is at most one. The integral triangle inequality and
monotonicity of the integral of $|K_n-K|$ give
\eqref{eq:area-primitive-lone-bounds}; the right sides do not depend on $x$.

The same estimates against zero give
$|f_{K_n}|,|f_{1,K_n}|\leq2+2B$ on the physical interval.
The three fixed functions $r,r',r''$ are bounded there by compactness.
Choose a common nonnegative bound $M$ for all six entries
$f_{K_n},f_{1,K_n},K_n,r,r',r''$.
For the full density
\[
 D(f,f_1,K,r,r_1,r_2)
 =f(fr_2+2f_1r_1-Kr)^2-Kf_1r(fr_1+f_1r/2)+K^2fr^2,
\]
the triangle and product inequalities give
\[
 |D|\leq M(M^2+2M^2+M^2)^2
       +M^3(M^2+M^2/2)+M^2MM^2.
\]
This constant is integrable on $[0,1]$. Every approximating integrand is
continuous and measurable. At almost every point the two primitives and
curvature converge, while the original probe and its derivatives remain
fixed. Polynomial continuity therefore gives convergence of the entire
density, and dominated convergence gives the stated action limit.
No derivative convergence of the curvature has been used.
\end{proof}

\begin{lemma}\label{lem:strict-critical-contraction}
Fix the smooth negative area probe $r$ from
Theorem~\ref{thm:critical-plateau-instability}, and set
\begin{equation}\label{eq:critical-curvature-contraction}
 \begin{aligned}
 K_t(x)&=(1-t)K_*(x)+2t,\\
 \kappa_t&=(1-t)L_*+2t,\qquad
 \Lambda_t=(1-t)L_*C_{\mathrm{CK}}+2t.
 \end{aligned}
\end{equation}
There is $0<t<1$ such that $A_{K_t}(r)<0$.
For this same $t$, the curvature is continuous on $\R$, has both exact
normalized moments, is symmetric on $[0,1]$ and has the same two hemisphere
order laws. Its box and its ratio satisfy
\begin{equation}\label{eq:strict-contracted-box}
 \begin{gathered}
 0<\kappa_t<2<\Lambda_t,\qquad
 \kappa_t\leq K_t(x)\leq\Lambda_t\quad(x\in\R),\\
 C_{\mathrm{CK}}-\frac{\Lambda_t}{\kappa_t}
 =\frac{2t(C_{\mathrm{CK}}-1)}{\kappa_t}>0.
 \end{gathered}
\end{equation}
The probe $r$, including its derivatives and endpoint data, is unchanged.
\end{lemma}

\begin{proof}
Both moments are affine in curvature; the constant curvature two has
moments two and one. A nonnegative coefficient $1-t$ preserves the
physical reflection and order laws and the original global box.
The strict bounds follow from $L_*<2<L_*C_{\mathrm{CK}}$.
Multiplying the ratio difference by the positive denominator $\kappa_t$
cancels the two copies of $(1-t)L_*C_{\mathrm{CK}}$, giving exactly
\eqref{eq:strict-contracted-box}.

For the actual primitives, linearity of the integral gives
\[
 f_{K_t}=(1-t)f_{K_*}+t f_2,\qquad
 f_{1,K_t}=(1-t)f_{1,K_*}+t f_{1,2}.
\]
Substitute these into the complete density with the fixed $r,r',r''$.
The resulting function of $(t,x)$ is continuous, so its integral over the
compact physical interval is continuous in $t$. Its value at zero is
strictly negative. A sufficiently small positive $t<1$ therefore retains
negative action. The ratio deficit is obtained before any smoothing.
\end{proof}

\begin{theorem}\label{thm:smooth-subcritical-negative-witness}
There are $0<t<1$, functions $K,r:\R\to\R$ smooth on $\R$, produced
sphere data $D$, and globally smooth maps $X,R:\R\to\R$ with the following
properties. The same area probe $r$ equals one near both endpoints, has
$r(0)=1$, $r'(0)=0$, and is nonconstant on $[0,1]$.
On $[0,1]$, the recovered $K$ is symmetric with the two hemisphere order
laws and satisfies
\[
 \kappa_t\leq K\leq\Lambda_t,\qquad
 \int_0^1K=2,\qquad \int_0^1xK(x)\,dx=1,\qquad A_K(r)<0,
\]
where the identical contracted box obeys \eqref{eq:strict-contracted-box}.
All endpoint, interval-mapping, inverse and actual probe-jet laws of
Theorem~\ref{thm:fully-tied-area-realization} hold for these same $K,r,X,D,R$.
In particular $R=r\circ X$ globally, and for every $x\in[0,1]$,
\[
 D.a(V_K(x))=\frac1{K(x)},\qquad R(V_K(x))=r(x).
\]
For every sphere point $p$ one has $K_{g_D}(p)=K(X(z(p)))$.
The original smooth one-form $h=M_D(R)$ is nonzero and satisfies
\begin{equation}\label{eq:strict-negative-geometric-witness}
 \begin{gathered}
 g_D\in\mathcal G_{\mathrm{rot}},\qquad
 \kappa_t\leq K_{g_D}\leq\Lambda_t,\qquad
 \frac{\Lambda_t}{\kappa_t}<C_{\mathrm{CK}},\\
 Q_{g_D}(h)=2\pi A_K(r)<0.
 \end{gathered}
\end{equation}
\end{theorem}

\begin{proof}
Choose the fixed probe and a positive contraction parameter from
Lemma~\ref{lem:strict-critical-contraction}.
Apply Theorem~\ref{thm:symmetric-exact-moment-recovery} with target mean two
and exactly the box $[\kappa_t,\Lambda_t]$.
Reindex by its almost-everywhere convergent subsequence. The strictly
increasing index tends to infinity, so the reindexed sequence still
converges in $L^1$ and retains every moment, shape and box condition.
Lemma~\ref{lem:complete-area-action-convergence}, applied to the same $r$,
gives convergence of the complete actions to the already negative
contracted action. Choose an actual negative member and call it $K$.

This $K$ is smooth on $\R$, positive on $[0,1]$ and has both exact moments.
Theorem~\ref{thm:fully-tied-area-realization} therefore constructs the
stated $X,D,R$ and retains all of their equations for the same $K,r$.
The original action identity and $2\pi>0$ give strict negativity.
Since the zero form has zero action, the original form is nonzero.
The physical range of $X$ places $X(z(p))$ in $[0,1]$ for every sphere point,
so the global curvature composition transports the identical box.
The identity diffeomorphism places $g_D$ in the represented class.
The strict ratio is the one already proved before recovery.
\end{proof}

\begin{lemma}\label{lem:constant-curvature-square-identity}
Let $(\Sigma,g)$ be a compact smooth Riemannian surface without boundary
whose actual Gauss curvature is the constant $\kappa\in\R$.
For every smooth one-form $h$,
\begin{equation}\label{eq:constant-curvature-squares}
 Q_g(h)=\int_\Sigma
 \left(|\Delta^\nabla h+\kappa h|_g^2
       +\kappa|\nabla h|_g^2+2\kappa|A_g h|_g^2\right)\,d\mu_g.
\end{equation}
Every term is integrable. If $\kappa\geq0$, then $Q_g(h)\geq0$ for every
smooth one-form, with no conformal-Killing or invariance condition.
\end{lemma}

\begin{proof}
The actual smooth tensors and their inner products have integrable norms
on the compact surface. Canonical Green integration gives
\[
 \int_\Sigma\langle h,\Delta^\nabla h\rangle_g\,d\mu_g
 =-\int_\Sigma|\nabla h|_g^2\,d\mu_g.
\]
Expanding the first square in \eqref{eq:constant-curvature-squares}, its
cross term contributes $-2\kappa\int|\nabla h|^2$.
Combining the displayed positive derivative term leaves exactly the
$-\kappa|\nabla h|^2$ term in the original four-term definition
\eqref{eq:complete-dissipation}. The other two terms are unchanged.
This proves the identity; when $\kappa\geq0$ its right-hand density is
nonnegative pointwise.
\end{proof}

Define $\mathcal S_{\mathrm{rot}}$ to consist of all real $C\geq1$ such
that $Q_g(h)\geq0$ for every $g\in\mathcal G_{\mathrm{rot}}$, every
positive actual curvature box
$\kappa>0$, $\kappa\leq K_g\leq\Lambda$ with $\Lambda/\kappa\leq C$,
and every original smooth one-form $h$. Put
$C_{\mathrm{rot}}=\sup\mathcal S_{\mathrm{rot}}$.

\begin{theorem}\label{thm:structural-strict-separation}
The unrestricted safe-cap set contains one, is downward closed among
caps at least one, and is bounded above. For a contraction parameter
$0<t<1$ supplied by the smooth negative witness,
\begin{equation}\label{eq:structural-strict-separation}
 1\leq C_{\mathrm{rot}}\leq\frac{\Lambda_t}{\kappa_t}
 <C_{\mathrm{CK}}=\sup\mathcal S_{\mathrm{CK}}.
\end{equation}
Thus the unrestricted rotational threshold is strictly smaller than the
actual universal conformal-Killing threshold.
\end{theorem}

\begin{proof}
A positive curvature box with ratio at most one satisfies
$\Lambda\leq\kappa$. Its pointwise bounds therefore force $K_g=\kappa$.
Lemma~\ref{lem:constant-curvature-square-identity} proves safety for every
smooth form, so $1\in\mathcal S_{\mathrm{rot}}$.
Downward closure follows directly from the universal box quantifier.

Use the actual smooth witness of
Theorem~\ref{thm:smooth-subcritical-negative-witness} and put
$b=\Lambda_t/\kappa_t$. If a safe cap $C$ were at least $b$, its universal
safety assertion would apply to that same metric, positive box and
nonzero original form, contradicting its negative action.
Consequently every safe cap is strictly below $b$.
The safe set is nonempty and bounded above; the real supremum laws give
$1\leq C_{\mathrm{rot}}\leq b$.
Its strict separation from the actual CK supremum now follows from
\eqref{eq:strict-contracted-box} and Theorem~\ref{thm:universal-ck-threshold}.
\end{proof}

The exact value of $C_{\mathrm{rot}}$ and smooth attainment at its
unrestricted endpoint are not determined here.
